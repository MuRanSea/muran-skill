9.14.6.15.2 ListBucketCustomDomain（Rust SDK）
list_bucket_custom_domain 接口用于获取指定 Bucket 的自定义域名配置列表，本文介绍如何通过 RustSDK 的 list_bucket_custom_domain 接口查询已绑定的域名、证书状态及禁用原因。
注意事项
您需要根据实际情况，填写示例代码中的地域和访问域名，详细介绍，请参见 地域和访问域名。为了防止密钥泄露，本文示例代码需要从环境变量中读取 Access Key ID 和 Secret Access Key。因此使用 SDK 前，您需要配置长期访问凭证，详细介绍，请参见 配置长期访问凭证。
权限说明
火山引擎主账号默认拥有全部权限，但是 IAM 用户默认没有任何权限。如果您需要使用子账号执行list_bucket_custom_domain 操作，火山引擎主账号需通过 IAM策略和桶策略为该子账号授予以下权限：tos:ListBucketCustomDomain

 }

 {

 }

#[tokio::main]
async fn main() {
// 初始化 SDK 客户端
let ak = env::var("TOS_ACCESS_KEY").unwrap_or_else(|_| "".to_string());
let sk = env::var("TOS_SECRET_KEY").unwrap_or_else(|_| "".to_string());
let endpoint = "https://tos-cn-beijing.volces.com";
let region = "cn-beijing";
let client = tos::builder::<TokioRuntime>()
 .connection_timeout(3000)
 .request_timeout(10000)
 .max_retry_count(3)
 .ak(ak)
 .sk(sk)
 .region(region)
 .endpoint(endpoint)
 .build()
 .unwrap();
// 创建请求输入
let input = ListBucketCustomDomainInput::new("test-bucket");
// 调用接口
match client.list_bucket_custom_domain(&input).await {
Ok(output) => {
println!("成功获取自定义域名列表");
for domain in output.rules() {
println!("域名: {}", domain.domain());
println!("证书ID: {}", domain.cert_id());
println!("证书状态: {}", domain.cert_status());
println!("协议类型: {}", domain.protocol());
println!("是否禁用: {}", domain.forbidden());
println!("禁用原因: {}", domain.forbidden_reason());
 }
 }
Err(err) => {
eprintln!("获取自定义域名列表失败: {:?}", err);

 }
 }
