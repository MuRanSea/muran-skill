9.2.8.4.2 查询工作流（Java SDK）
本文介绍如何通过 TOS Java SDK 查询工作流。
注意事项
仅 Java SDK 2.9.8 及以上版本支持查询工作流的功能。
示例代码
// 查询转码工作流
package org.example.GetConvertWorkflow;
import com.volcengine.tos.TOSV2;
import com.volcengine.tos.TOSV2ClientBuilder;
import com.volcengine.tos.TosClientException;
import com.volcengine.tos.TosServerException;
import com.volcengine.tos.comm.common.*;
import com.volcengine.tos.model.bucket.*;
import com.volcengine.tos.model.object.*;
public class GetConvertWorkflowExample {
public static void main(String[] args) {
String endpoint = System.getenv("TOS_ENDPOINT");
String region = System.getenv("TOS_REGION");
String accessKey = System.getenv("TOS_ACCESS_KEY");
String secretKey = System.getenv("TOS_SECRET_KEY");
// 填写 BucketName
String bucketName = System.getenv("TOS_BUCKET");
// 如果未设置区域，默认使用华北2（北京）区域
if (region == null || region.isEmpty()) {
 region = "cn-beijing";
 }
// 初始化客户端
TOSV2 tos = new TOSV2ClientBuilder().build(region, endpoint, accessKey, secretKey);try {
// 构造 GetConvertWorkflowInput，查询指定桶下配置的转码工作流
GetConvertWorkflowInput input = GetConvertWorkflowInput.builder()
 .bucket(bucketName)
 .build();
// 调用 getConvertWorkflow 接口
GetConvertWorkflowOutput output = tos.getConvertWorkflow(input);
// 打印请求信息
if (output != null && output.getRequestInfo() != null) {
 System.out.println("RequestID: " + output.getRequestInfo().getRequestId());
 System.out.println("StatusCode: " + output.getRequestInfo().getStatusCode());
 }
// 打印规则信息
if (output != null && output.getRules() != null) {
 System.out.println("规则数量: " + output.getRules().size());
if (!output.getRules().isEmpty()) {
ConvertWorkflowRule firstRule = output.getRules().get(0);
if (firstRule != null) {
// 规则基本信息
 System.out.println("首条规则ID: " + firstRule.getId());
 System.out.println("首条规则Enabled: " + firstRule.getEnabled());
 System.out.println("首条规则Prefix: " + firstRule.getPrefix());
// 音视频扩展过滤配置
WorkflowExtFilter extFilter = firstRule.getExtFilter();
if (extFilter != null) {
if (extFilter.getAudioExts() != null) {
 System.out.println("音频扩展 AudioExts: " + extFilter.getAudioExts());
 }
if (extFilter.getVideoExts() != null) {
 System.out.println("视频扩展 VideoExts: " + extFilter.getVideoExts());
 }
 }
// 拓扑信息
if (firstRule.getTopology() != null) {
 System.out.println("拓扑 Topology: " + firstRule.getTopology());
 }
// 操作配置
WorkflowOperations operations = firstRule.getOperations();
if (operations != null) {
// Transcode 操作
if (operations.getTranscode() != null && !operations.getTranscode().isEmpty()) {OperationsTranscode transcodeOp = operations.getTranscode().get(0);
if (transcodeOp != null) {
 System.out.println("首个Transcode OperationID: " + 
transcodeOp.getOperationID());
 System.out.println("首个Transcode TemplateID: " + transcodeOp.getTemplateID());ConvertJobOutput transcodeOutput = transcodeOp.getOutput();
if (transcodeOutput != null) {
 System.out.println("首个Transcode Output Region: " + 
transcodeOutput.getRegion());
 System.out.println("首个Transcode Output Bucket: " + 
transcodeOutput.getBucket());
 System.out.println("首个Transcode Output Object: " + 
transcodeOutput.getObject());
 }
 }
 }
// AudioTranscode 操作
if (operations.getAudioTranscode() != null && !
operations.getAudioTranscode().isEmpty()) {
OperationsAudioTranscode audioTranscodeOp =
operations.getAudioTranscode().get(0);
if (audioTranscodeOp != null) {
 System.out.println("首个AudioTranscode OperationID: " + 
audioTranscodeOp.getOperationID());
 System.out.println("首个AudioTranscode TemplateID: " + 
audioTranscodeOp.getTemplateID());
ConvertJobOutput audioOutput = audioTranscodeOp.getOutput();
if (audioOutput != null) {
 System.out.println("首个AudioTranscode Output Region: " + 
audioOutput.getRegion());
 System.out.println("首个AudioTranscode Output Bucket: " + 
audioOutput.getBucket());
 System.out.println("首个AudioTranscode Output Object: " + 
audioOutput.getObject());
 }
 }
 }
 }
 }
 }
 }
 } catch (TosClientException e) {
// 操作失败，捕获客户端异常，一般情况是请求参数错误，此时请求并未发送
 System.out.println("操作失败");
 System.out.println("Message: " + e.getMessage());
if (e.getCause() != null) {
 e.getCause().printStackTrace();
 }
 } catch (TosServerException e) {
// 操作失败，捕获服务端异常，可以获取到从服务端返回的详细错误信息
**相关文档 **
关于查询工作流的相关 API 接口，请参见 GetWorkflow 与 GetWorkflowExecution 。
 System.out.println("操作失败");
 System.out.println("StatusCode: " + e.getStatusCode());
 System.out.println("Code: " + e.getCode());
 System.out.println("Message: " + e.getMessage());
 System.out.println("RequestID: " + e.getRequestID());
 } catch (Throwable t) {
// 作为兜底捕获其他异常，一般不会执行到这里
 System.out.println("操作失败");
 System.out.println("unexpected exception, message: " + t.getMessage());
 }
 }
}
