9.2.8.3.2 列举音频转码模板（JavaSDK））
本文介绍如何通过 TOS Java SDK 列举音频转码模板。
注意事项
仅 Java SDK 2.9.8 及以上版本支持列举音频转码模板的功能。
示例代码
// 列举音频转码模板
import com.volcengine.tos.TOSV2;
import com.volcengine.tos.TOSV2ClientBuilder;
import com.volcengine.tos.TosClientException;
import com.volcengine.tos.TosServerException;
import com.volcengine.tos.comm.common.AudioContainerFormatType;
import com.volcengine.tos.model.bucket.*;
public class ListAudioConvertTemplatesExample {
public static void main(String[] args) {
String endpoint = System.getenv("TOS_ENDPOINT");
String region = System.getenv("TOS_REGION");
String accessKey = System.getenv("TOS_ACCESS_KEY");
String secretKey = System.getenv("TOS_SECRET_KEY");
// 填写 BucketName
String bucketName = System.getenv("TOS_BUCKET");
// 初始化客户端
TOSV2 tos = new TOSV2ClientBuilder().build(region, endpoint, accessKey, secretKey);try {
// 创建列举音频转码模板请求
ListAudioConvertTemplatesInput input = ListAudioConvertTemplatesInput.builder()
 .bucket(bucketName)
 .build();
// 调用列举音频转码模板接口
ListAudioConvertTemplatesOutput output = tos.listAudioConvertTemplates(input);
// 处理返回结果
 System.out.println("列举音频转码模板成功");
 System.out.println("RequestID: " + output.getRequestInfo().getRequestId());
 System.out.println("StatusCode: " + output.getRequestInfo().getStatusCode());
// 获取音频转码模板列表
 java.util.List<AudioConvertTemplate> templates = output.getAudioConvertTemplates();if (templates != null && !templates.isEmpty()) {
 System.out.println("找到 " + templates.size() + " 个音频转码模板:");
for (AudioConvertTemplate template : templates) {
 System.out.println("\n模板信息:");
 System.out.println(" ID: " + template.getId());
 System.out.println(" 名称: " + template.getName());
 System.out.println(" 标签: " + template.getTag());
// 打印音频转码配置信息
AudioConvertConfig config = template.getAudioConvertConfig();
if (config != null) {
 System.out.println(" 音频配置:");
 System.out.println(" 容器格式: " + config.getContainerFormat());
 System.out.println(" 比特率: " + config.getBitRate());
 System.out.println(" 采样率: " + config.getSampleRate());
 System.out.println(" 声道数: " + config.getChannels());
 System.out.println(" 采样格式: " + config.getSampleFormat());
// 打印时间区间信息
TimeInterval timeInterval = config.getTimeInterval();
if (timeInterval != null) {
 System.out.println(" 时间区间:");
 System.out.println(" 起始时间: " + timeInterval.getStart());
 System.out.println(" 持续时间: " + timeInterval.getDuration());
 }
 }
 }
 } else {
 System.out.println("未找到音频转码模板");
 }
 } catch (TosClientException e) {
// 操作失败，捕获客户端异常，一般情况是请求参数错误，此时请求并未发送
 System.out.println("操作失败");
 System.out.println("Message: " + e.getMessage());
if (e.getCause() != null) {
 e.getCause().printStackTrace();
 }
 } catch (TosServerException e) {
// 操作失败，捕获服务端异常，可以获取到从服务端返回的详细错误信息
 System.out.println("操作失败");
 System.out.println("StatusCode: " + e.getStatusCode());
 System.out.println("Code: " + e.getCode());
 System.out.println("Message: " + e.getMessage());
 System.out.println("RequestID: " + e.getRequestID());
**相关文档 **
关于创建音频转码模版的相关API接口，请参见ListAudioConvertTemplates。
 } catch (Throwable t) {
// 作为兜底捕获其他异常，一般不会执行到这里
 System.out.println("操作失败");
 System.out.println("unexpected exception, message: " + t.getMessage());
 }
 }
}
