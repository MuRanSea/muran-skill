9.14.7 对象接口
9.14.7.1 基础操作
9.14.7.1.1 CopyObject（Rust SDK）
copy_object 接口用于在同一地域下同一桶或不同桶之间拷贝对象，本文介绍如何通过 Rust SDK 的copy_object 接口完成对象拷贝，支持元数据替换、服务端加密及版本控制。
注意事项
您需要根据实际情况，填写示例代码中的地域和访问域名，详细介绍，请参见 地域和访问域名。为了防止密钥泄露，本文示例代码需要从环境变量中读取 Access Key ID 和 Secret Access Key。因此使用 SDK 前，您需要配置长期访问凭证，详细介绍，请参见 配置长期访问凭证。
当源对象大小超过 5 GiB 时，请改用 upload_part_copy 接口，copy_object 仅支持 0-5 GiB 范围。若源对象属于 ARCHIVE、COLD_ARCHIVE 或 DEEP_COLD_ARCHIVE 存储类型，需先执行恢复操作，再调用 copy_object。
权限说明
火山引擎主账号默认拥有全部权限，但是 IAM 用户默认没有任何权限。如果您需要使用子账号执行copy_object 操作，火山引擎主账号需通过 IAM策略和桶策略为该子账号授予以下权限：
tos:PutObject
代码示例
同一桶内复制对象
use async_trait::async_trait;
use futures_core::future::BoxFuture;
use std::env;
use std::future::Future;
use std::time::Duration;
use tokio::runtime::Handle;
use ve_tos_rust_sdk::common::init_tracing_log;
use ve_tos_rust_sdk::asynchronous::bucket::BucketAPI;
use ve_tos_rust_sdk::asynchronous::object::ObjectAPI;
use ve_tos_rust_sdk::asynchronous::tos;
use ve_tos_rust_sdk::asynchronous::tos::AsyncRuntime;
use ve_tos_rust_sdk::object::{CopyObjectInput, PutObjectFromBufferInput};
use ve_tos_rust_sdk::error::TosError;
#[derive(Debug, Default)]
pub struct TokioRuntime {}
#[async_trait]
impl AsyncRuntime for TokioRuntime {
type JoinError = tokio::task::JoinError;
async fn sleep(&self, duration: Duration) {
 tokio::time::sleep(duration).await;
 }
fn spawn<'a, F>(&self, future: F) -> BoxFuture<'a, Result<F::Output, Self::JoinError>>where
 F: Future + Send + 'static,
 F::Output: Send + 'static,
 {
Box::pin(Handle::current().spawn(future))
 }
fn block_on<F: Future>(&self, future: F) -> F::Output {
 Handle::current().block_on(future)
 }
}
#[tokio::main]
async fn main() {
let _guard = init_tracing_log("info", "temp/logs", "app.log");
let ak = env::var("TOS_ACCESS_KEY").unwrap_or("".to_string());
let sk = env::var("TOS_SECRET_KEY").unwrap_or("".to_string());
let endpoint = "https://tos-cn-beijing.volces.com";
let region = "cn-beijing";
let client = tos::builder::<TokioRuntime>()
 .connection_timeout(3000)
 .request_timeout(10000)
 .max_retry_count(3)
 .ak(ak)
 .sk(sk)
 .region(region)
 .endpoint(endpoint)
 .build().unwrap();
// Upload source object
let source_key = "source-object.txt";
// Same bucket copy
let copy_input = CopyObjectInput::new("example-bucket", "copied-object.txt", "example-bucket", source_key);
let output = client.copy_object(&copy_input).await;
match output {
Ok(res) => {
复制对象并设置元数据
println!("Copy object succeeded");
println!("etag: {}", res.etag());
if let Some(lm) = res.last_modified() { println!("last_modified: {}", lm); }
println!("copy_source_version_id: {}", res.copy_source_version_id());
println!("version_id: {}", res.version_id());
 }
Err(err) => {
match err {
 TosError::TosClientError { message, .. } => println!("Client error: {}", message),
 TosError::TosServerError { request_id, ec, code, .. } => println!("Server error: request_id: {}, ec: 
{}, code: {}", request_id, ec, code),
 _ => println!("Unknown error: {:?}", err)
 }
 }
 }
}
