6.5.3.5.5 ListVectors（TOS Vectors Node.js SDK）
ListVectors 接口用于列举向量，本文介绍如何通过 TOS Vectors Node.js SDK 的 ListVectors 接口完成该操作。
注意事项
您需要根据实际情况，填写示例代码中的地域和访问域名，详细介绍，请参见地域和访问域名（Endpoint）。
为了防止密钥泄露，本文示例代码需要从环境变量中读取 Access Key ID 和 Secret Access Key。因此使用 SDK 前，您需要配置长期访问凭证，详细介绍，请参见配置长期访问凭证。
TOS Vectors Node.js SDK 支持使用临时访问凭证；也支持配置超时机制、重试策略等多种机制，您可以根据业务需求灵活设置，提升应用安全性，详细介绍，请参见常见配置场景。
权限说明
主账号默认拥有全部权限，但是 IAM 用户默认没有任何权限。如果您需要使用 IAM 用户执行 ListVectors操作，主账号需通过 IAM策略和桶策略为该 IAM 用户授予以下权限： tosvectors:ListVectors代码示例
列举向量基本用法
// 列举向量基本用法
import {
TosVectorsClient,
TosClientError,
TosServerError,
} from "@volcengine/tos-sdk";
// 从环境变量中获取认证信息和配置
const client = new TosVectorsClient({
accessKeyId: process.env["TOS_ACCESS_KEY"], // TOS_ACCESS_KEY: 您的 Access Key IDaccessKeySecret: process.env["TOS_SECRET_KEY"], // TOS_SECRET_KEY: 您的 Secret Keyregion: 'Provide your region',
endpoint: 'Provide your vector endpoint',
});
/**
 * 处理 SDK 调用过程中可能出现的错误。
 * @param error 捕获到的错误对象
 */
function handleError(error: Error) {
if (error instanceof TosClientError) {
// 客户端异常，通常是参数错误或网络问题
console.log("Client Error:");
console.log(" Message:", error.message);
if (error.stack) {
console.log(" Stack:", error.stack);
 }
 } else if (error instanceof TosServerError) {
// 服务端异常，表示请求已送达但处理失败
console.log("Server Error:");
console.log(" Request ID:", error.requestId);
console.log(" Status Code:", error.statusCode);
console.log(" Code:", error.code);
console.log(" Message:", error.message);
console.log(" Headers:", error.headers);
 } else {
// 其他未知异常
console.log("Unexpected Error:");
console.log(" Message:", error.message);
 }
}
async function main() {
const vectorBucketName = 'Provide your vector bucket name';
const accountId = process.env["TOS_ACCOUNT_ID"];
const indexName = "test";
try {
console.log(
`列举向量桶 ${vectorBucketName} 中索引 ${indexName} 下的向量列表...`,
 );
// 1. 调用 listVectors 方法，列举索引下的向量列表，并返回数据和元数据
const result = await client.listVectors({
 vectorBucketName,
accountID: accountId,
 indexName,
returnData: true,
returnMetadata: true,
 });
// 2. 打印通用返回信息
console.log("Request Succeeded:");
console.log(" Request ID:", result.requestId);
console.log(" Status Code:", result.statusCode);
console.log(" Headers:", result.headers);
// 3. 打印业务数据：返回的向量数量以及每个向量的 key、向量维度和元数据字段
const vectors = result.data.vectors || [];
console.log(` 返回向量数量: ${vectors.length}`);
分页列举向量
if (vectors.length === 0) {
console.log(" 当前请求未返回任何向量。");
 }
 vectors.forEach((vector, index) => {
const dataLength = vector.data?.float32?.length ?? 0;
const metadataKeys = vector.metadata ? Object.keys(vector.metadata) : [];
console.log(` 向量 #${index + 1}:`);
console.log(` - Key: ${vector.key}`);
if (dataLength > 0) {
console.log(` - 向量维度（float32 长度）: ${dataLength}`);
 } else {
console.log(" - 向量数据: <未返回或为空>");
 }
if (metadataKeys.length > 0) {
console.log(` - 元数据字段: ${metadataKeys.join(", ")}`);
 } else {
console.log(" - 元数据字段: <无>");
 }
 });
// 4. 打印下一页标记（如存在）
if (result.data.nextToken) {
console.log("Next Token:", result.data.nextToken);
 }
 } catch (error: any) {
handleError(error);
 }
}
main();

async function main() {
const vectorBucketName = 'Provide your vector bucket name';
const accountId = process.env["TOS_ACCOUNT_ID"];
const indexName = "test";
let maxResults = 100;
try {
console.log("开始分页列举向量...");
let nextToken: string | undefined = undefined;
let page = 1;
// 使用 nextToken 进行分页循环
do {
console.log(`\n请求第 ${page} 页向量列表...`);
const result = await client.listVectors({
 vectorBucketName,
accountID: accountId,
 indexName,
 maxResults,
 nextToken,
 });
// 每页打印请求通用信息
console.log("Page Request Info:");
console.log(" Request ID:", result.requestId);
console.log(" Status Code:", result.statusCode);
console.log(" Headers:", result.headers);
// 打印本页向量列表摘要
const vectors = result.data.vectors || [];
console.log(` 本页返回向量数量: ${vectors.length}`);
if (vectors.length === 0) {
console.log(" 本页未返回任何向量。");
 }
 vectors.forEach((vector, index) => {
const dataLength = vector.data?.float32?.length ?? 0;
const metadataKeys = vector.metadata ? Object.keys(vector.metadata) : [];
console.log(` 向量 #${index + 1}:`);
console.log(` - Key: ${vector.key}`);
if (dataLength > 0) {
console.log(` - 向量维度（float32 长度）: ${dataLength}`);
 } else {
console.log(" - 向量数据: <未返回或为空>");
 }
if (metadataKeys.length > 0) {
console.log(` - 元数据字段: ${metadataKeys.join(", ")}`);
 } else {
console.log(" - 元数据字段: <无>");
 }
 });
 nextToken = result.data.nextToken;
if (nextToken) {
console.log(" Next Token:", nextToken);
 page += 1;
 } else {
console.log(" 已无更多分页结果。");
 }
 } while (nextToken);
 } catch (error: any) {
handleError(error);
 }
}
main();
