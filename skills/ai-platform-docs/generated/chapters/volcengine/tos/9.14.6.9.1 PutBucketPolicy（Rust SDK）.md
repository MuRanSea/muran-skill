9.14.6.9 桶策略
9.14.6.9.1 PutBucketPolicy（Rust SDK）
put_bucket_policy 接口用于修改或创建桶策略，若已存在则覆盖。本文介绍如何通过 Rust SDK 的put_bucket_policy 接口设置桶策略，需桶所有者或授权子账号调用，策略总大小不超过 20 KB。注意事项
您需要根据实际情况，填写示例代码中的地域和访问域名，详细介绍，请参见 地域和访问域名。为了防止密钥泄露，本文示例代码需要从环境变量中读取 Access Key ID 和 Secret Access Key。因此使用 SDK 前，您需要配置长期访问凭证，详细介绍，请参见 配置长期访问凭证。
桶策略 JSON 序列化后总大小不可超过 20 KB。
新策略会完全覆盖已有策略，原有内容将被替换。
策略条数无限制，但须确保 JSON 格式合法且符合语法规范。
权限说明
火山引擎主账号默认拥有全部权限，但是 IAM 用户默认没有任何权限。如果您需要使用子账号执行put_bucket_policy 操作，火山引擎主账号需通过 IAM策略和桶策略为该子账号授予以下权限：tos:PutBucketPolicy
代码示例
设置存储桶策略
use async_trait::async_trait;
use futures_core::future::BoxFuture;
use std::env;
use std::future::Future;
use std::time::Duration;
use tokio::runtime::Handle;
use ve_tos_rust_sdk::common::init_tracing_log;
use ve_tos_rust_sdk::asynchronous::bucket::BucketAPI;
use ve_tos_rust_sdk::asynchronous::tos;
use ve_tos_rust_sdk::asynchronous::tos::AsyncRuntime;
use ve_tos_rust_sdk::bucket::PutBucketPolicyInput;
use ve_tos_rust_sdk::error::TosError;
#[derive(Debug, Default)]
pub struct TokioRuntime {}
#[async_trait]
impl AsyncRuntime for TokioRuntime {
type JoinError = tokio::task::JoinError;
async fn sleep(&self, duration: Duration) {
 tokio::time::sleep(duration).await;
 }
fn spawn<'a, F>(&self, future: F) -> BoxFuture<'a, Result<F::Output, Self::JoinError>>where
 F: Future + Send + 'static,
 F::Output: Send + 'static,
 {
Box::pin(Handle::current().spawn(future))
 }
fn block_on<F: Future>(&self, future: F) -> F::Output {
 Handle::current().block_on(future)
 }
}
#[tokio::main]
async fn main() {
let _guard = init_tracing_log("info", "temp/logs", "app.log");
let ak = env::var("TOS_ACCESS_KEY").unwrap_or(""
 .to_string());
let sk = env::var("TOS_SECRET_KEY").unwrap_or(""
 .to_string());
let endpoint = "https://tos-cn-beijing.volces.com";
let region = "cn-beijing";
let client = tos::builder::<TokioRuntime>()
 .connection_timeout(3000)
 .request_timeout(10000)
 .max_retry_count(3)
 .ak(ak)
 .sk(sk)
 .region(region)
 .endpoint(endpoint)
 .build().unwrap();
// 创建存储桶策略输入
let bucket_name = "my-bucket";
let policy = r#"{
 "Version": "2012-10-17",
 "Statement": [
 {
 "Sid": "桶只读策略",
 "Effect": "Allow",
 "Principal": ["210004xxxx"],
 "Action": ["tos:Get*", "tos:List*"],
相关文档
关于 PutBucketPolicy 的 API 文档，请参见 PutBucketPolicy。
 "Resource": ["trn:tos:::my-bucket", "trn:tos:::my-bucket/*"]
 }
 ]
 }"#;
let mut input = PutBucketPolicyInput::new(bucket_name);
 input.set_policy(policy);
// 调用设置存储桶策略接口
let output = client.put_bucket_policy(&input).await;
match output {
Ok(output) => {
let request_id = output.request_id();
println!("设置存储桶策略成功，Request ID: {}", request_id);
// 检查请求ID长度
assert!(request_id.len() > 0, "请求ID不能为空");
 }
Err(err) => {
match err {
 TosError::TosClientError { message, .. } => {
println!("客户端错误: {}", message);
 }
 TosError::TosServerError { request_id, ec, .. } => {
println!("服务端错误 - Request ID: {}, 错误码: {}", request_id, ec);
 }
 }
 }
 }
}
