9.14.6.4.2 GetBucketACL（Rust SDK）
get_bucket_acl 接口用于获取指定桶的 ACL 信息，本文介绍如何通过 Rust SDK 的 get_bucket_acl 接口读取桶访问控制列表。
注意事项
您需要根据实际情况，填写示例代码中的地域和访问域名，详细介绍，请参见 地域和访问域名。为了防止密钥泄露，本文示例代码需要从环境变量中读取 Access Key ID 和 Secret Access Key。因此使用 SDK 前，您需要配置长期访问凭证，详细介绍，请参见 配置长期访问凭证。
调用者必须拥有 READ_ACP 权限，否则请求返回 403 Forbidden。
响应中 BucketAclDelivered 字段仅反映桶级 ACL 继承开关状态，不表示对象当前 ACL 实际值。若桶未配置任何自定义授权，Grants 数组仅返回所有者 FULL_CONTROL 记录。
权限说明
火山引擎主账号默认拥有全部权限，但是 IAM 用户默认没有任何权限。如果您需要使用子账号执行get_bucket_acl 操作，火山引擎主账号需通过 IAM策略和桶策略为该子账号授予以下权限：tos:GetBucketACL
代码示例
获取指定桶的访问控制权限
use async_trait::async_trait; 
use futures_core::future::BoxFuture;
use std::env;
use std::future::Future;
use std::time::Duration;
use tokio::runtime::Handle;
use ve_tos_rust_sdk::common::init_tracing_log;
use ve_tos_rust_sdk::asynchronous::bucket::BucketAPI;
use ve_tos_rust_sdk::asynchronous::tos;
use ve_tos_rust_sdk::asynchronous::tos::AsyncRuntime;
use ve_tos_rust_sdk::bucket::GetBucketACLInput;
use ve_tos_rust_sdk::error::TosError;
#[derive(Debug, Default)]
pub struct TokioRuntime {}
#[async_trait]
impl AsyncRuntime for TokioRuntime {
type JoinError = tokio::task::JoinError;
async fn sleep(&self, duration: Duration) {
 tokio::time::sleep(duration).await;
 }
fn spawn<'a, F>(&self, future: F) -> BoxFuture<'a, Result<F::Output, Self::JoinError>>where
 F: Future + Send + 'static,
 F::Output: Send + 'static,
 {
Box::pin(Handle::current().spawn(future))
 }
fn block_on<F: Future>(&self, future: F) -> F::Output {
 Handle::current().block_on(future)
 }
}
#[tokio::main]
async fn main() {
// 初始化日志
let _guard = init_tracing_log("info", "temp/logs", "app.log");
// 从环境变量获取AK/SK
let ak = env::var("TOS_ACCESS_KEY").unwrap_or("".to_string());
let sk = env::var("TOS_SECRET_KEY").unwrap_or("".to_string());
let endpoint = "https://tos-cn-beijing.volces.com";
let region = "cn-beijing";
// 构建TOS客户端
let client = tos::builder::<TokioRuntime>()
 .connection_timeout(3000)
 .request_timeout(10000)
 .max_retry_count(3)
 .ak(ak)
 .sk(sk)
 .region(region)
 .endpoint(endpoint)
 .build().unwrap();
// 用例1: 获取指定桶的访问控制权限
let bucket_name = "example-bucket";
let input = GetBucketACLInput::new(bucket_name);
let output = client.get_bucket_acl(&input).await;
match output {
Ok(output) => {
println!("请求ID: {}", output.request_id());
println!("桶所有者ID: {}", output.owner().id());
println!("ACL继承状态: {}", output.bucket_acl_delivered());
相关文档
关于 GetBucketACL 的 API 文档，请参见 GetBucketACL。
println!("访问控制权限列表:");
for grant in output.grants() {
println!(" 被授权者类型: {:?}", grant.grantee().grantee_type());
println!(" 权限: {:?}", grant.permission());
if let Some(canned) = grant.grantee().canned() {
println!(" 预定义组: {:?}", canned);
 }
 }
 }
Err(err) => {
match err {
 TosError::TosClientError { message, .. } => {
println!("客户端错误: {}", message);
 }
 TosError::TosServerError { request_id, ec, .. } => {
println!("服务端错误 - 请求ID: {}, 错误码: {}", request_id, ec);
 }
 }
 }
 }
}
