6.5.2.3.2 ListVectorBuckets（TOS Vectors Java SDK）
ListVectorBuckets 接口用于列举向量桶，本文介绍如何通过 TOS Vectors Java SDK 的 ListVectorBuckets 接口列举向量桶并获取桶元数据。
注意事项
您需要根据实际情况，填写示例代码中的地域和访问域名，详细介绍，请参见 地域和访问域名（Endpoint）。
为了防止密钥泄露，本文示例代码需要从环境变量中读取 Access Key ID 和 Secret Access Key。因此使用 SDK 前，您需要配置长期访问凭证，详细介绍，请参见 配置长期访问凭证。
TOS Vectors Java SDK 支持使用临时访问凭证；也支持配置超时机制、重试策略等多种机制，您可以根据业务需求灵活设置，提升应用安全性，详细介绍，请参见 常见配置场景。
权限说明
主账号默认拥有全部权限，但是 IAM 用户默认没有任何权限。如果您需要使用 IAM 用户执行ListVectorBuckets 操作，主账号需通过 IAM策略和桶策略为该 IAM 用户授予以下权限：
tosvectors:ListVectorBuckets

// 列举向量桶基本用法
import com.volcengine.tos.TOSVectorsClientBuilder;
import com.volcengine.tos.TOSVectors;
import com.volcengine.tos.TosClientException;
import com.volcengine.tos.TosServerException;
import com.volcengine.tos.model.vectors.ListVectorBucketsInput;
import com.volcengine.tos.model.vectors.ListVectorBucketsOutput;
import com.volcengine.tos.model.vectors.VectorBucket;
public class ListVectorBucketsExample {
public static void main(String[] args) {
String endpoint = "Provide your endpoint";
String region = "Provide your region";
String accessKey = System.getenv("TOS_ACCESS_KEY");
String secretKey = System.getenv("TOS_SECRET_KEY");
// 初始化客户端
TOSVectors client = new TOSVectorsClientBuilder().build(region, endpoint, accessKey, secretKey);try {
// 创建列举向量桶请求
ListVectorBucketsInput input = ListVectorBucketsInput.builder().build();
分页列举向量桶
// 调用列举向量桶接口
ListVectorBucketsOutput output = client.listVectorBuckets(input);
 System.out.println("RequestID: " + output.getRequestInfo().getRequestId());
 System.out.println("StatusCode: " + output.getRequestInfo().getStatusCode());
// 打印向量桶列表信息
if (output.getVectorBuckets() != null && !output.getVectorBuckets().isEmpty()) {
 System.out.println("向量桶数量: " + output.getVectorBuckets().size());
for (VectorBucket bucket : output.getVectorBuckets()) {
 System.out.println("向量桶名称: " + bucket.getVectorBucketName());
 System.out.println("项目名称: " + bucket.getProjectName());

 }
 } else {
 System.out.println("当前没有向量桶");

// 分页列举向量桶
import com.volcengine.tos.TOSVectorsClientBuilder;
import com.volcengine.tos.TOSVectors;
import com.volcengine.tos.TosClientException;
import com.volcengine.tos.TosServerException;
import com.volcengine.tos.model.vectors.ListVectorBucketsInput;
import com.volcengine.tos.model.vectors.ListVectorBucketsOutput;
import com.volcengine.tos.model.vectors.VectorBucket;
import java.util.ArrayList;
import java.util.List;
public class ListVectorBucketsExample2 {
public static void main(String[] args) {
String endpoint = 'Provide your endpoint';
String region = 'Provide your region';
String accessKey = System.getenv("TOS_ACCESS_KEY");
String secretKey = System.getenv("TOS_SECRET_KEY");
// 初始化客户端
TOSVectors client = new TOSVectorsClientBuilder().build(region, endpoint, accessKey, secretKey);try {
// 分页列举向量桶，每页显示2个
int maxResultsPerPage = 2;
String nextToken = null;
int pageCount = 0;
int totalBucketCount = 0;
do {
 pageCount++;
// 构建分页请求
 ListVectorBucketsInput.Builder inputBuilder = ListVectorBucketsInput.builder()
 .maxResults(maxResultsPerPage);
// 如果不是第一页，设置nextToken
if (nextToken != null) {
 inputBuilder.nextToken(nextToken);
 }
ListVectorBucketsInput input = inputBuilder.build();
// 调用列举向量桶接口
ListVectorBucketsOutput output = client.listVectorBuckets(input);
 System.out.println("第 " + pageCount + " 页:");
 System.out.println("RequestID: " + output.getRequestInfo().getRequestId());
 System.out.println("StatusCode: " + output.getRequestInfo().getStatusCode());
// 处理当前页的向量桶
if (output.getVectorBuckets() != null && !output.getVectorBuckets().isEmpty()) {int currentPageCount = output.getVectorBuckets().size();
 totalBucketCount += currentPageCount;
 System.out.println("当前页向量桶数量: " + currentPageCount);
for (VectorBucket bucket : output.getVectorBuckets()) {
 System.out.println(" 向量桶名称: " + bucket.getVectorBucketName());
 System.out.println(" 项目名称: " + bucket.getProjectName());
 System.out.println(" 创建时间: " + bucket.getCreationTime());
 }
 } else {
 System.out.println("当前页没有向量桶");
 }
// 获取下一页的token
 nextToken = output.getNextToken();
 System.out.println("是否有下一页: " + (nextToken != null ? "是" : "否"));
 System.out.println("---");
 } while (nextToken != null); // 当存在nextToken时继续分页
 System.out.println("分页列举完成!");
 System.out.println("总页数: " + pageCount);
 System.out.println("总向量桶数量: " + totalBucketCount);
 } catch (TosClientException e) {
// 操作失败，捕获客户端异常，一般情况是请求参数错误，此时请求并未发送
 System.out.println("操作失败");
 System.out.println("Message: " + e.getMessage());
if (e.getCause() != null) {
 e.getCause().printStackTrace();
 }
 } catch (TosServerException e) {
// 操作失败，捕获服务端异常，可以获取到从服务端返回的详细错误信息
 System.out.println("操作失败");
 System.out.println("StatusCode: " + e.getStatusCode());
 System.out.println("Code: " + e.getCode());
 System.out.println("Message: " + e.getMessage());
 System.out.println("RequestID: " + e.getRequestID());
 } catch (Throwable t) {
// 作为兜底捕获其他异常，一般不会执行到这里
 System.out.println("操作失败");
 System.out.println("unexpected exception, message: " + t.getMessage());
 }
 }
}
