9.14.7.1.5 HeadObject（Rust SDK）
head_object 接口用于获取对象元数据而不返回对象本体，本文介绍如何通过 Rust SDK 的 head_object 接口完成该操作，并处理 SSE-C 等加密场景。
注意事项
您需要根据实际情况，填写示例代码中的地域和访问域名，详细介绍，请参见 地域和访问域名。为了防止密钥泄露，本文示例代码需要从环境变量中读取 Access Key ID 和 Secret Access Key。因此使用 SDK 前，您需要配置长期访问凭证，详细介绍，请参见 配置长期访问凭证。
若对象采用 SSE-TOS 或 SSE-KMS 加密，head_object 请求中禁止携带 x_tos_server_side_encryption 相关头域，否则返回 400 BadRequest。
若对象采用 SSE-C 加密，head_object 必须携带 x_tos_server_side_encryption_customer_algorithm 、x_tos_server_side_encryption_customer_key 、 x_tos_server_side_encryption_customer_key_md5 ，三者缺一不可。
对软链接对象， ETag 、 x_tos_storage_class 、 x_tos_hash_crc64ecma 、 x_tos_symlink_target_size 返回目标对象值， Last-Modified 取软链接与目标对象较晚时间，其余头域返回软链接自身值。
请求头带 version_id 参数时，head_object 返回指定版本对象的元数据；若版本不存在则响应 404 NotFound。
权限说明
火山引擎主账号默认拥有全部权限，但是 IAM 用户默认没有任何权限。如果您需要使用子账号执行head_object 操作，火山引擎主账号需通过 IAM策略和桶策略为该子账号授予以下权限：
tos:GetObject

use async_trait::async_trait;
use futures_core::future::BoxFuture;
use std::env;
use std::future::Future;
use std::time::Duration;
use tokio::runtime::Handle;
use ve_tos_rust_sdk::common::init_tracing_log;
use ve_tos_rust_sdk::asynchronous::object::ObjectAPI;
use ve_tos_rust_sdk::asynchronous::tos;
use ve_tos_rust_sdk::asynchronous::tos::AsyncRuntime;
use ve_tos_rust_sdk::object::HeadObjectInput;
use ve_tos_rust_sdk::error::TosError;
#[derive(Debug, Default)]
pub struct TokioRuntime {}
#[async_trait]
impl AsyncRuntime for TokioRuntime {
type JoinError = tokio::task::JoinError;
async fn sleep(&self, duration: Duration) {
 tokio::time::sleep(duration).await;
 }
fn spawn<'a, F>(&self, future: F) -> BoxFuture<'a, Result<F::Output, Self::JoinError>>where
 F: Future + Send + 'static,
 F::Output: Send + 'static,
 {
Box::pin(Handle::current().spawn(future))
 }
fn block_on<F: Future>(&self, future: F) -> F::Output {
 Handle::current().block_on(future)
 }
}
#[tokio::main]
async fn main() {
let _guard = init_tracing_log("info", "temp/logs", "app.log");
let ak = env::var("TOS_ACCESS_KEY").unwrap_or("".to_string());
let sk = env::var("TOS_SECRET_KEY").unwrap_or("".to_string());
let endpoint = "https://tos-cn-beijing.volces.com";
let region = "cn-beijing";
let client = tos::builder::<TokioRuntime>()
 .connection_timeout(3000)
 .request_timeout(10000)
 .max_retry_count(3)
 .ak(ak)
 .sk(sk)
 .region(region)
 .endpoint(endpoint)
 .build().unwrap();
let input = HeadObjectInput::new("example-bucket", "example-object.txt");
let output = client.head_object(&input).await;
match output {
Ok(output) => {
println!("etag: {}", output.etag());
if let Some(last_modified) = output.last_modified() {
println!("last_modified: {}", last_modified);
 }
println!("delete_marker: {}", output.delete_marker());
println!("version_id: {}", output.version_id());
