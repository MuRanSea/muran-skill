9.2.6.3.2 查询 ObjectSet 存储用量（Java SDK）
本文介绍如何通过 TOS Java SDK 查询指定 ObjectSet 的实际存储用量与对象数，并按存储类型维度返回统计值。
注意事项
ObjectSet 必须存在。
通过子账号调用时，需要具备当前 Bucket 的 tos:GetObjectSetStorage 权限。
用量统计可能存在服务端统计延迟，请以服务端返回结果为准。
代码示例
import com.volcengine.tos.TOSV2;
import com.volcengine.tos.TOSV2ClientBuilder;
import com.volcengine.tos.TosClientException;
import com.volcengine.tos.TosServerException;
import com.volcengine.tos.model.bucket.GetObjectSetStorageInput;
import com.volcengine.tos.model.bucket.GetObjectSetStorageOutput;
import com.volcengine.tos.model.bucket.ObjectSetStorageStat;
public class GetObjectSetStorageExample {
public static void main(String[] args) {
String endpoint = System.getenv("TOS_ENDPOINT");
String region = System.getenv("TOS_REGION");
String accessKey = System.getenv("TOS_ACCESS_KEY");
String secretKey = System.getenv("TOS_SECRET_KEY");
String bucket = System.getenv("TOS_BUCKET");
String objectSetName = System.getenv("TOS_OBJECT_SET_NAME");
TOSV2 tos = new TOSV2ClientBuilder().build(region, endpoint, accessKey, secretKey);try {
GetObjectSetStorageInput input = GetObjectSetStorageInput.builder()
 .bucket(bucket)
 .objectSetName(objectSetName)
 .build();
GetObjectSetStorageOutput output = tos.getObjectSetStorage(input);
 System.out.println("getObjectSetStorage succeed");
ObjectSetStorageStat total = output.getTotalStorageStat();
if (total != null) {
 System.out.println("TotalStorage: " + total.getStorage());
 System.out.println("TotalObjectCount: " + total.getObjectCount());
 }
 System.out.println("RequestID: " + output.getRequestInfo().getRequestId());
相关文档
关于查询 ObjectSet 存储用量的相关 API 接口，请参见 GetObjectSetStorage 。
 } catch (TosClientException e) {
 System.out.println("getObjectSetStorage failed");
 System.out.println("Message: " + e.getMessage());
if (e.getCause() != null) {
 e.getCause().printStackTrace();
 }
 } catch (TosServerException e) {
 System.out.println("getObjectSetStorage failed");
 System.out.println("StatusCode: " + e.getStatusCode());
 System.out.println("Code: " + e.getCode());
 System.out.println("Message: " + e.getMessage());
 System.out.println("RequestID: " + e.getRequestID());
 } catch (Throwable t) {
 System.out.println("getObjectSetStorage failed");
 System.out.println("unexpected exception, message: " + t.getMessage());
 }
 }
}
