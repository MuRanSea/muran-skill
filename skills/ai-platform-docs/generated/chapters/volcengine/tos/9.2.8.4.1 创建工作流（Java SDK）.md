9.2.8.4 工作流
9.2.8.4.1 创建工作流（Java SDK）
本文介绍如何通过 TOS Java SDK 创建工作流。
注意事项
仅 Java SDK 2.9.8 及以上版本支持创建工作流的功能。
示例代码
创建视频转码工作流
// 创建视频转码工作流
package org.example.PutConvertWorkflow;
import com.volcengine.tos.TOSV2;
import com.volcengine.tos.TOSV2ClientBuilder;
import com.volcengine.tos.TosClientException;
import com.volcengine.tos.TosServerException;
import com.volcengine.tos.comm.common.*;
import com.volcengine.tos.model.bucket.*;
import com.volcengine.tos.model.object.*;
public class PutConvertWorkflowExample {
public static void main(String[] args) {
String endpoint = System.getenv("TOS_ENDPOINT");
String region = System.getenv("TOS_REGION");
String accessKey = System.getenv("TOS_ACCESS_KEY");
String secretKey = System.getenv("TOS_SECRET_KEY");
// 填写 BucketName
String bucketName = System.getenv("TOS_BUCKET");
// 初始化客户端
TOSV2 tos = new TOSV2ClientBuilder().build(region, endpoint, accessKey, secretKey);try {
// 场景说明：创建一个视频转码工作流，当指定 Bucket 下 videos/ 前缀中的 mp4 文件上传时，自动触发转码任务
// 1. 创建视频转码模板，配置容器、视频、音频以及时间区间等参数
// 设置时间区间：从第 0ms 开始，持续 60s
TimeInterval timeInterval = TimeInterval.builder()
 .start(0)
 .duration(60_000)
 .build();
// 配置输出容器：使用 MP4 格式，并设置一个较短的剪辑时长（示例为 10s）
ClipConfig clipConfig = ClipConfig.builder()
// 片段时长，单位毫秒
 .duration(10_000)
 .build();
Container container = Container.builder()
 .format(ContainerFormatType.MP4)
 .clipConfig(clipConfig)
 .build();
// 配置视频编码参数，这里使用常见的 H.264 编码
Video video = Video.builder()
 .codec("h264")
 .width(1920)
 .height(1080)// - 分辨率：1920x1080
 .crf(23) // - CRF：23
 .pixFmt("yuv420p") // - 像素格式：yuv420p
 .bitRate(4_000_000) // - 视频码率：4 Mbps
 .fps(25) // - 帧率：25 fps
 .remove(false)
 .build();
// 配置音频编码参数，这里使用 AAC 编码
Audio audio = Audio.builder()
 .codec("aac")
 .bitRate(128_000) // - 码率：128 kbps
 .sampleFormat("fltp")
 .sampleRate(44_100) // - 采样率：44100 Hz
 .channels(2) // - 声道数：2（立体声）
 .remove(false)
 .build();
// 汇总视频转码配置
Transcode transcode = Transcode.builder()
 .timeInterval(timeInterval)
 .container(container)
 .video(video)
 .audio(audio)
 .build();
// 创建视频转码模板，模板名称可根据业务自行命名
PutVideoConvertTemplateInput putVideoTemplateInput =
PutVideoConvertTemplateInput.builder()
 .bucket(bucketName)
 .name("example-video-convert-workflow-template")
 .transcodeConfig(transcode)
 .build();
PutVideoConvertTemplateOutput putVideoTemplateOutput =
tos.putVideoConvertTemplate(putVideoTemplateInput);
// 2. 构造转码工作流中的转码操作
// 配置转码输出位置，这里示例写入到同一 Bucket 的 output/video.mp4
ConvertJobOutput videoOutput = ConvertJobOutput.builder()
 .region(region)
 .bucket(bucketName)
 .object("output/video.mp4")
 .build();
// operationID 用于在拓扑中引用该操作，这里命名为 "transcode"
OperationsTranscode videoTranscodeOperation = OperationsTranscode.builder()
 .operationID("transcode")
 .templateID(putVideoTemplateOutput.getId())
 .output(videoOutput)
 .build();
// WorkflowOperations 中仅包含视频转码操作
WorkflowOperations workflowOperations = WorkflowOperations.builder()
 .transcode(java.util.Arrays.asList(videoTranscodeOperation))
 .build();
// 3. 设置扩展名过滤器，仅对 mp4 后缀的视频文件生效
WorkflowExtFilter extFilter = WorkflowExtFilter.builder()
 .videoExts(java.util.Arrays.asList("mp4"))
 .build();
// 4. 构造转码工作流规则
// 规则 ID 建议做到全局唯一，这里简单使用当前时间戳拼接前缀
String ruleId = "video-convert-workflow-" + System.currentTimeMillis();
// 拓扑配置：只有一个节点，表示执行 transcode 操作
 java.util.List<java.util.List<String>> topology = java.util.Arrays.asList(
 java.util.Arrays.asList("transcode")
 );
ConvertWorkflowRule rule = ConvertWorkflowRule.builder()
 .id(ruleId)
// 规则开启
 .enabled(true)
// 只匹配 videos/ 前缀下的对象
 .prefix("videos/")
 .extFilter(extFilter)
创建音频转码工作流
 .topology(topology)
 .operations(workflowOperations)
 .build();
// 5. 调用 putConvertWorkflow 接口创建工作流
PutConvertWorkflowInput putWorkflowInput = PutConvertWorkflowInput.builder()
 .bucket(bucketName)
 .rules(java.util.Arrays.asList(rule))
 .build();
PutConvertWorkflowOutput putWorkflowOutput =
tos.putConvertWorkflow(putWorkflowInput);
// 打印关键返回字段及规则概要信息
 System.out.println("创建视频转码工作流成功");
 System.out.println("RequestID: " + putWorkflowOutput.getRequestInfo().getRequestId());
 System.out.println("StatusCode: " + putWorkflowOutput.getRequestInfo().getStatusCode());
 System.out.println("规则 ID: " + ruleId);
 System.out.println("视频转码输出位置: " + videoOutput.getBucket() + "/" + 
videoOutput.getObject());
 } catch (TosClientException e) {
// 操作失败，捕获客户端异常，一般情况是请求参数错误，此时请求并未发送
 System.out.println("操作失败");
 System.out.println("Message: " + e.getMessage());
if (e.getCause() != null) {
 e.getCause().printStackTrace();
 }
 } catch (TosServerException e) {
// 操作失败，捕获服务端异常，可以获取到从服务端返回的详细错误信息
 System.out.println("操作失败");
 System.out.println("StatusCode: " + e.getStatusCode());
 System.out.println("Code: " + e.getCode());
 System.out.println("Message: " + e.getMessage());
 System.out.println("RequestID: " + e.getRequestID());
 } catch (Throwable t) {
// 作为兜底捕获其他异常，一般不会执行到这里
 System.out.println("操作失败");
 System.out.println("unexpected exception, message: " + t.getMessage());
 }
 }
}

// 创建音频转码工作流
package org.example.PutConvertWorkflow;
import com.volcengine.tos.TOSV2;
import com.volcengine.tos.TOSV2ClientBuilder;
import com.volcengine.tos.TosClientException;
import com.volcengine.tos.TosServerException;
import com.volcengine.tos.comm.common.*;
import com.volcengine.tos.model.bucket.*;
import com.volcengine.tos.model.object.*;
public class PutConvertWorkflowExample2 {
public static void main(String[] args) {
String endpoint = System.getenv("TOS_ENDPOINT");
String region = System.getenv("TOS_REGION");
String accessKey = System.getenv("TOS_ACCESS_KEY");
String secretKey = System.getenv("TOS_SECRET_KEY");
// 填写 BucketName
String bucketName = System.getenv("TOS_BUCKET");
// 初始化客户端
TOSV2 tos = new TOSV2ClientBuilder().build(region, endpoint, accessKey, secretKey);try {
// 场景说明：创建一个音频转码工作流，当指定 Bucket 下 audios/ 前缀中的 mp3 文件上传时，自动触发音频转码任务
// 1. 创建音频转码模板，配置容器格式、比特率、采样率、声道数以及时间区间
// 设置时间区间：从第 0ms 开始，持续 5 分钟
TimeInterval timeInterval = TimeInterval.builder()
 .start(0)
 .duration(300_000)
 .build();
// 配置音频转码参数
AudioConvertConfig audioConvertConfig = AudioConvertConfig.builder()
 .timeInterval(timeInterval)
// - 容器格式：MP3
 .containerFormat(AudioContainerFormatType.MP3)
// - 码率：128 kbps
 .bitRate(128_000)
 .bitRateOpt(1)
// - 采样格式：fltp
 .sampleFormat("fltp")
// - 采样率：44100 Hz
 .sampleRate(44_100)
// - 声道数：2（立体声）
 .channels(2)
 .build();
// 创建音频转码模板
PutAudioConvertTemplateInput putAudioTemplateInput =
PutAudioConvertTemplateInput.builder()
 .bucket(bucketName)
 .name("example-audio-convert-workflow-template")
 .audioConvertConfig(audioConvertConfig)
 .build();
PutAudioConvertTemplateOutput putAudioTemplateOutput =
tos.putAudioConvertTemplate(putAudioTemplateInput);
// 2. 构造转码工作流中的音频转码操作
// 配置转码输出位置，这里示例写入到同一 Bucket 的 output/audio.mp3
ConvertJobOutput audioOutput = ConvertJobOutput.builder()
 .region(region)
 .bucket(bucketName)
 .object("output/audio.mp3")
 .build();
// operationID 用于在拓扑中引用该操作，这里命名为 "audio-transcode"
OperationsAudioTranscode audioTranscodeOperation = OperationsAudioTranscode.builder()
 .operationID("audio-transcode")
 .templateID(putAudioTemplateOutput.getId())
 .output(audioOutput)
 .build();
// WorkflowOperations 中仅包含音频转码操作
WorkflowOperations workflowOperations = WorkflowOperations.builder()
 .audioTranscode(java.util.Arrays.asList(audioTranscodeOperation))
 .build();
// 3. 设置扩展名过滤器，仅对 mp3 音频文件生效
WorkflowExtFilter extFilter = WorkflowExtFilter.builder()
 .audioExts(java.util.Arrays.asList("mp3"))
 .build();
// 4. 构造音频转码工作流规则
String ruleId = "audio-convert-workflow-" + System.currentTimeMillis();
// 拓扑配置：只有一个节点，表示执行 audio-transcode 操作
 java.util.List<java.util.List<String>> topology = java.util.Arrays.asList(
 java.util.Arrays.asList("audio-transcode")
 );
ConvertWorkflowRule rule = ConvertWorkflowRule.builder()
 .id(ruleId)
// 规则开启
 .enabled(true)
// 只匹配 audios/ 前缀下的对象
 .prefix("audios/")
 .extFilter(extFilter)
 .topology(topology)
 .operations(workflowOperations)
 .build();
// 5. 调用 putConvertWorkflow 接口创建音频转码工作流
PutConvertWorkflowInput putWorkflowInput = PutConvertWorkflowInput.builder()
 .bucket(bucketName)
 .rules(java.util.Arrays.asList(rule))
 .build();
PutConvertWorkflowOutput putWorkflowOutput =
tos.putConvertWorkflow(putWorkflowInput);
// 打印关键返回字段及规则概要信息
 System.out.println("创建音频转码工作流成功");
 System.out.println("RequestID: " + putWorkflowOutput.getRequestInfo().getRequestId());
 System.out.println("StatusCode: " + putWorkflowOutput.getRequestInfo().getStatusCode());
 System.out.println("规则 ID: " + ruleId);
 System.out.println("音频转码输出位置: " + audioOutput.getBucket() + "/" + 
audioOutput.getObject());
 } catch (TosClientException e) {
// 操作失败，捕获客户端异常，一般情况是请求参数错误，此时请求并未发送
 System.out.println("操作失败");
 System.out.println("Message: " + e.getMessage());
if (e.getCause() != null) {
 e.getCause().printStackTrace();
 }
 } catch (TosServerException e) {
// 操作失败，捕获服务端异常，可以获取到从服务端返回的详细错误信息
 System.out.println("操作失败");
 System.out.println("StatusCode: " + e.getStatusCode());
 System.out.println("Code: " + e.getCode());
 System.out.println("Message: " + e.getMessage());
 System.out.println("RequestID: " + e.getRequestID());
 } catch (Throwable t) {
// 作为兜底捕获其他异常，一般不会执行到这里
 System.out.println("操作失败");
 System.out.println("unexpected exception, message: " + t.getMessage());
 }
