12.11 使用 AWS S3 SDK 访问 TOS
本文旨在帮助您快速、正确地配置各主流编程语言的 AWS S3 SDK，以连接并操作 TOS。
概述
TOS 提供了与 Amazon S3 兼容的 API 接口。这意味着您可以利用 AWS 提供功能丰富、生态完善的 S3SDK，通过简单的配置，无缝地访问和管理 TOS 上的存储资源。
统一的技术栈：在多云环境中，使用同一套 AWS S3 SDK 管理不同云厂商的存储服务，简化开发和维护工作。
平滑迁移：对于从 AWS S3 或其他兼容 S3 的存储服务迁移至 TOS 的应用，几乎无需更改业务逻辑代码，只需修改客户端配置即可完成切换。
利用成熟生态：享受 AWS S3 SDK 带来的稳定、高效的开发体验，以及社区中丰富的示例和问题解决方案。
前提条件
已创建扁平桶，桶的分类及创建请参考存储桶分类和创建存储桶。
SDK 的签名版本推荐配置为 v4 （ s3v4 ），以保证连接安全性。
火山引擎 TOS 兼容 S3 的域名仅支持虚拟主机样式（ Virtual-hosted Style ） ，不支持路径样式（ Pathstyle ）。因此，在配置 SDK 时，必须强制开启或默认使用虚拟主机样式，如果 SDK 默认为 Pathstyle，则需要显式关闭它。这是确保连接成功的关键。参考信息
地域（Region）：您的 TOS 存储桶所在的地域，例如 cn-beijing 。详情请查看地域和访问域名（Endpoint）。
访问域名（Endpoint）：可选参数，详情请查看地域和访问域名（Endpoint）。
访问密钥（Access Key）：包含 AK（Access Key ID) 和 SK（Secret Access Key)，是访问 TOS API 的身份凭证。创建和管理访问密钥详情请参见 API 访问密钥（Access Key） 。

import software.amazon.awssdk.auth.credentials.AwsBasicCredentials;
import software.amazon.awssdk.auth.credentials.StaticCredentialsProvider;
import software.amazon.awssdk.core.ResponseBytes;
import software.amazon.awssdk.core.sync.RequestBody;
import software.amazon.awssdk.core.sync.ResponseTransformer;
import software.amazon.awssdk.regions.Region;
import software.amazon.awssdk.services.s3.S3Client;
import software.amazon.awssdk.services.s3.model.*;
import java.io.IOException;
import java.net.URI;
import java.nio.file.Files;
import java.nio.file.Paths;
public class TosS3JavaBestPractice {
public static void main(String[] args) {
// 1. 配置认证信息和访问端点
final String endpoint = "https://tos-s3-cn-beijing.volces.com";
final String region = "cn-beijing";
final String accessKey = System.getenv("VOLCENGINE_ACCESS_KEY_ID");
final String secretKey = System.getenv("VOLCENGINE_SECRET_ACCESS_KEY");
// 2. 初始化 S3 客户端
// 注意：AWS SDK for Java v2 默认使用虚拟主机样式（Virtual-hosted Style），
// 如果需要显式关闭 Path-style（某些旧版本或特殊配置可能需要），可使
用 .forcePathStyle(false)
S3Client s3Client = S3Client.builder()
 .endpointOverride(URI.create(endpoint))
 .region(Region.of(region))
 .credentialsProvider(StaticCredentialsProvider.create(
 AwsBasicCredentials.create(accessKey, secretKey)))
 .build();
String bucketName = "your-unique-bucket-name-" + System.currentTimeMillis();
String objectKey = "my-object-key";
String localFilePath = "local-file.txt";
String downloadFilePath = "downloaded-file.txt";
try {
// 准备本地文件
 Files.write(Paths.get(localFilePath), "Hello, Volcengine TOS!".getBytes());
// 3. 创建存储桶
 System.out.println("Creating bucket: " + bucketName);
 s3Client.createBucket(CreateBucketRequest.builder().bucket(bucketName).build());
 System.out.println("Bucket created successfully.");
// 4. 上传对象
 System.out.println("Uploading object to " + bucketName);
 s3Client.putObject(PutObjectRequest.builder()
 .bucket(bucketName)
 .key(objectKey)
 .build(), RequestBody.fromFile(Paths.get(localFilePath)));
 System.out.println("Object uploaded successfully.");
// 5. 列举对象
 System.out.println("Listing objects in bucket: " + bucketName);
ListObjectsV2Response listObjectsV2Response =
s3Client.listObjectsV2(ListObjectsV2Request.builder()
 .bucket(bucketName).build());
for (S3Object s3Object : listObjectsV2Response.contents()) {
 System.out.println("- " + s3Object.key());
 }
// 6. 下载对象
 System.out.println("Downloading object: " + objectKey);
ResponseBytes<GetObjectResponse> objectBytes = s3Client.getObject(
 GetObjectRequest.builder().bucket(bucketName).key(objectKey).build(),
 ResponseTransformer.toBytes());
 Files.write(Paths.get(downloadFilePath), objectBytes.asByteArray());
 System.out.println("Object downloaded successfully to " + downloadFilePath);
// 7. 删除对象
 System.out.println("Deleting object: " + objectKey);
s3Client.deleteObject(DeleteObjectRequest.builder().bucket(bucketName).key(objectKey).build());
 System.out.println("Object deleted successfully.");
// 8. 删除存储桶
 System.out.println("Deleting bucket: " + bucketName);
 s3Client.deleteBucket(DeleteBucketRequest.builder().bucket(bucketName).build());
 System.out.println("Bucket deleted successfully.");
 } catch (S3Exception | IOException e) {
 System.err.println("Error: " + e.getMessage());
 e.printStackTrace();
 } finally {
// 清理本地文件
try {
 Files.deleteIfExists(Paths.get(localFilePath));
 Files.deleteIfExists(Paths.get(downloadFilePath));
 } catch (IOException ex) {
 System.err.println("Failed to clean up local files: " + ex.getMessage());
 }
 s3Client.close();
 }

import boto3
from botocore.config import Config
from botocore.exceptions import ClientError
import os
import time
def main():
# 1. 配置认证信息和访问端点
 endpoint = "https://tos-s3-cn-beijing.volces.com"
 region = "cn-beijing"
 access_key = os.getenv("VOLCENGINE_ACCESS_KEY_ID")
 secret_key = os.getenv("VOLCENGINE_SECRET_ACCESS_KEY")
# 2. 初始化 S3 客户端
# 关键配置：
# - endpoint_url: 指向 TOS S3 域名
# - region_name: 必须提供，即使 endpoint_url 已指定
# - config:
# - signature_version: 's3v4'
# - s3: {'addressing_style': 'virtual'} 强制使用虚拟主机寻址模式
 s3_client = boto3.client(
's3',
 endpoint_url=endpoint,
 aws_access_key_id=access_key,
 aws_secret_access_key=secret_key,
 region_name=region,
 config=Config(
 signature_version='s3v4',
 s3={'addressing_style': 'virtual'} # 明确指定虚拟主机模式
 )
 )
 bucket_name = f"your-unique-bucket-name-{int(time.time())}"
 object_key = "my-object-key"
 local_file_path = "local-file.txt"
 download_file_path = "downloaded-file.txt"
try:
# 准备本地文件
with open(local_file_path, "w") as f:
 f.write("Hello, Volcengine TOS!")
# 3. 创建存储桶
print(f"Creating bucket: {bucket_name}")
# 对于 cn-beijing 地域，不需要指定 LocationConstraint
 s3_client.create_bucket(Bucket=bucket_name)
print("Bucket created successfully.")
# 4. 上传对象
print(f"Uploading object to {bucket_name}")
 s3_client.upload_file(local_file_path, bucket_name, object_key)
print("Object uploaded successfully.")
# 5. 列举对象
print(f"Listing objects in bucket: {bucket_name}")
 response = s3_client.list_objects_v2(Bucket=bucket_name)
if 'Contents' in response:
for obj in response['Contents']:
print(f"- {obj['Key']}")
# 6. 下载对象
print(f"Downloading object: {object_key}")
 s3_client.download_file(bucket_name, object_key, download_file_path)
print(f"Object downloaded successfully to {download_file_path}")
# 7. 删除对象
print(f"Deleting object: {object_key}")
 s3_client.delete_object(Bucket=bucket_name, Key=object_key)

常见问题
关于 CreateBucket 时的 LocationConstraint
当您在美国东部（弗吉尼亚北部) us-east-1 以外的地域创建存储桶时，S3 API 要求必须提供 LocationConstraint 。TOS 的 S3 兼容接口目前不需要此参数，即使在 cn-beijing 等非 us-east-1 的地域。您可以省略此配置。如果 SDK 强制要求，传入与 region 相同的值即可。
证书验证失败 (Certificate Validation Error) 
企业内网或代理环境中，可能会偶发 SSL 证书验证拦截。请确保您的开发环境信任 *.volces.com 域名的证书。
部分 SDK（如 Python boto3 ) 允许通过参数 verify=False 禁用 SSL 证书验证以进行调试，但这存在安全隐患。强烈建议您在生产环境中通过配置系统或 CA 证书信任链来彻底解决此问题。
签名版本不匹配 (Signature Version Mismatch)
调用接口时若返回签名验证相关错误，请检查您的 SDK 初始化配置，确保已强制指定使用 v4 签名算法（s3v4）， TOS 的 S3 兼容接口强依赖此签名版本。
