9.2.8.3 音频处理
9.2.8.3.1 创建音频转码模板（Java SDK）
本文介绍如何通过 TOS Java SDK 创建音频转码模板。
注意事项
仅 Java SDK 2.9.8 及以上版本支持创建音频转码模板功能。
代码示例
创建音频转码模板
// 创建音频转码模板
import com.volcengine.tos.TOSV2;
import com.volcengine.tos.TOSV2ClientBuilder;
import com.volcengine.tos.TosClientException;
import com.volcengine.tos.TosServerException;
import com.volcengine.tos.comm.common.AudioContainerFormatType;
import com.volcengine.tos.model.bucket.PutAudioConvertTemplateInput;
import com.volcengine.tos.model.bucket.PutAudioConvertTemplateOutput;
import com.volcengine.tos.model.bucket.AudioConvertConfig;
import com.volcengine.tos.model.bucket.TimeInterval;
public class PutAudioConvertTemplateExample {
public static void main(String[] args) {
String endpoint = System.getenv("TOS_ENDPOINT");
String region = System.getenv("TOS_REGION");
String accessKey = System.getenv("TOS_ACCESS_KEY");
String secretKey = System.getenv("TOS_SECRET_KEY");
// 填写 BucketName
String bucketName = System.getenv("TOS_BUCKET");
// 初始化客户端
TOSV2 tos = new TOSV2ClientBuilder().build(region, endpoint, accessKey, secretKey);try {
// 创建音频转码模板
// 1. 设置时间间隔参数：从第1秒开始，持续10秒
TimeInterval timeInterval = TimeInterval.builder()
 .start(1) // 开始时间（毫秒）
 .duration(10000) // 持续时间（毫秒）
 .build();
// 2. 配置音频转码参数
AudioConvertConfig audioConvertConfig = AudioConvertConfig.builder()
 .timeInterval(timeInterval) // 时间间隔配置
 .containerFormat(AudioContainerFormatType.MP3) // 容器格式：MP3
 .bitRate(8000) // 比特率：8000 bps
 .bitRateOpt(1)
 .sampleFormat("fltp") // 采样格式：fltp
 .sampleRate(11025) // 采样率：11025 Hz
 .channels(2) // 声道数：2（立体声）
 .build();
// 3. 创建音频转码模板输入参数
PutAudioConvertTemplateInput input = PutAudioConvertTemplateInput.builder()
 .bucket(bucketName)
 .name("test-audio-template") // 模板名称
 .audioConvertConfig(audioConvertConfig) // 音频转码配置
 .build();
// 4. 调用创建音频转码模板API
PutAudioConvertTemplateOutput output = tos.putAudioConvertTemplate(input);
// 5. 处理返回结果
 System.out.println("创建音频转码模板成功");
 System.out.println("RequestID: " + output.getRequestInfo().getRequestId());
 System.out.println("StatusCode: " + output.getRequestInfo().getStatusCode());
 System.out.println("模板ID: " + output.getId());
 System.out.println("模板名称: test-audio-template");
 System.out.println("容器格式: MP3");
 System.out.println("比特率: 8000 bps");
 System.out.println("采样率: 11025 Hz");
 System.out.println("声道数: 2");
 } catch (TosClientException e) {
// 操作失败，捕获客户端异常，一般情况是请求参数错误，此时请求并未发送
 System.out.println("操作失败");
 System.out.println("Message: " + e.getMessage());
if (e.getCause() != null) {
 e.getCause().printStackTrace();
 }
 } catch (TosServerException e) {
// 操作失败，捕获服务端异常，可以获取到从服务端返回的详细错误信息
 System.out.println("操作失败");
 System.out.println("StatusCode: " + e.getStatusCode());
 System.out.println("Code: " + e.getCode());
 System.out.println("Message: " + e.getMessage());
 System.out.println("RequestID: " + e.getRequestID());
相关文档
关于创建音频转码模版的相关API接口，请参见PutAudioConvertTemplate。
 } catch (Throwable t) {
// 作为兜底捕获其他异常，一般不会执行到这里
 System.out.println("操作失败");
 System.out.println("unexpected exception, message: " + t.getMessage());
 }
 }
}
