9.2.8.6 视频转动图
9.2.8.6.1 查询视频转动图任务（Java SDK）
本文介绍如何通过 TOS Java SDK 查询视频转动图任务。
注意事项
仅 Java SDK 2.9.15 及以上版本支持查询视频转动图任务的功能。
任务为异步执行，返回的状态参数 state 可能为 Submitted 、 Running 、 Success 、 Failed 、Paused 或 Canceled 。
示例代码
以下代码展示如何根据任务 ID 查询任务状态和输出对象信息。
import com.volcengine.tos.TOSV2;
import com.volcengine.tos.TOSV2ClientBuilder;
import com.volcengine.tos.TosClientException;
import com.volcengine.tos.TosServerException;
import com.volcengine.tos.model.bucket.AnimationConfig;
import com.volcengine.tos.model.bucket.GetAnimationJobInput;
import com.volcengine.tos.model.bucket.GetAnimationJobOutput;
public class GetAnimationJobExample {
public static void main(String[] args) {
String endpoint = System.getenv("TOS_ENDPOINT");
String region = System.getenv("TOS_REGION");
String accessKey = System.getenv("TOS_ACCESS_KEY");
String secretKey = System.getenv("TOS_SECRET_KEY");
String bucketName = System.getenv("TOS_BUCKET");
String objectKey = System.getenv("TOS_OBJECT_KEY");
String jobId = System.getenv("TOS_ANIMATION_JOB_ID");
TOSV2 tos = new TOSV2ClientBuilder().build(region, endpoint, accessKey, secretKey);try {
GetAnimationJobOutput output = tos.getAnimationJob(GetAnimationJobInput.builder()
 .bucket(bucketName)
 .key(objectKey)
 .jobID(jobId)
 .build());
 System.out.println("getAnimationJob succeed.");
 System.out.println("Job ID: " + output.getJobID());
 System.out.println("State: " + (output.getState() == null ? null : output.getState().getValue()));
 System.out.println("Message: " + output.getMessage());

if (output.getInput() != null) {
 System.out.println("Input object: " + output.getInput().getObject());
 }
if (output.getOutput() != null) {
 System.out.println("Output bucket: " + output.getOutput().getBucket());
 System.out.println("Output object: " + output.getOutput().getObject());
 }
AnimationConfig animationConfig = output.getAnimationConfig();
if (animationConfig != null && animationConfig.getAnimation() != null) {
 System.out.println("Animation width: " + animationConfig.getAnimation().getWidth());
 System.out.println("Animation height: " + animationConfig.getAnimation().getHeight());
 System.out.println("Animation fps: " + animationConfig.getAnimation().getFps());
 }
 System.out.println("Request ID: " + output.getRequestInfo().getRequestId());
 } catch (TosClientException e) {
 System.out.println("getAnimationJob failed");
 System.out.println("Message: " + e.getMessage());
if (e.getCause() != null) {
 e.getCause().printStackTrace();
 }
 } catch (TosServerException e) {
 System.out.println("getAnimationJob failed");
 System.out.println("StatusCode: " + e.getStatusCode());
 System.out.println("Code: " + e.getCode());
 System.out.println("Message: " + e.getMessage());
 System.out.println("RequestID: " + e.getRequestID());
 } catch (Throwable t) {
 System.out.println("getAnimationJob failed");
 System.out.println("unexpected exception, message: " + t.getMessage());
 }
 }
}
