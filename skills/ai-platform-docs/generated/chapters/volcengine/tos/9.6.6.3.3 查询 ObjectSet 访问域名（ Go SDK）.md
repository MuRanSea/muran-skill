9.6.6.3.3 查询 ObjectSet 访问域名（ Go SDK）
本文介绍如何通过 TOS Go SDK 查询指定 ObjectSet 在已启用 ObjectSet 规则的汇聚接入点下生成的访问域名，包括 TOS 协议域名和 S3 协议域名。
注意事项
ObjectSet 必须存在。
该接口仅返回 CAP ObjectSet 规则状态为 Enabled 的接入点。
如果 Bucket 没有满足条件的接入点规则，服务端可能返回未找到、无权限或参数相关错误。通过子账号调用时，需要具备当前 Bucket 的 tos:GetObjectSetEndPoint 权限。
示例代码
package main
import (
"context"
"fmt"
"os"
"github.com/volcengine/ve-tos-golang-sdk/v2/tos"
)
func checkErr(err error) {
if err != nil {
if serverErr, ok := err.(*tos.TosServerError); ok {
 fmt.Println("Error:", serverErr.Error())
 fmt.Println("Request ID:", serverErr.RequestID)
 fmt.Println("Response Status Code:", serverErr.StatusCode)
 fmt.Println("Response Header:", serverErr.Header)
 fmt.Println("Response Err Code:", serverErr.Code)
 fmt.Println("Response Err Msg:", serverErr.Message)
 } else if clientErr, ok := err.(*tos.TosClientError); ok {
 fmt.Println("Error:", clientErr.Error())
 fmt.Println("Client Cause Err:", clientErr.Cause.Error())
 } else {
 fmt.Println("Error:", err)
 }
panic(err)
 }
}

 accessKey = os.Getenv("TOS_ACCESS_KEY")
 secretKey = os.Getenv("TOS_SECRET_KEY")
// Bucket 对应的 Endpoint，以华北 2（北京）为例：https://tos-cn-beijing.volces.com
 endpoint = "https://tos-cn-beijing.volces.com"
 region = "cn-beijing"
// 填写 BucketName
 bucketName = "*** Provide your bucket name ***"
// 填写 ObjectSet 名称，需满足 Bucket 的 PathLevel 和分隔符配置
 objectSetName = "docs/object/set"
 ctx = context.Background()
 )
// 初始化客户端
 client, err := tos.NewClientV2(endpoint, tos.WithRegion(region), 
tos.WithCredentials(tos.NewStaticCredentials(accessKey, secretKey)))
 checkErr(err)
 getOutput, err := client.GetObjectSetEndpoint(ctx, &tos.GetObjectSetEndpointInput{
 Bucket: bucketName,
 ObjectSetName: objectSetName,
 })
 checkErr(err)
 fmt.Println("GetObjectSetEndpoint Request ID:", getOutput.RequestID)
for _, endpoint := range getOutput.Endpoints {
 fmt.Println("CapName:", endpoint.CapName)
 fmt.Println("TOS Endpoints:", endpoint.Endpoint)
 fmt.Println("S3 Endpoints:", endpoint.S3Endpoint)
 }
}
