9.2.8.3.5 查询音频转码任务（Java SDK）
本文介绍如何通过 TOS Java SDK 查询音频转码任务。
注意事项
仅 Java SDK 2.9.8 及以上版本支持查询音频转码任务的功能。
示例代码
// 查询音频转码任务
package org.example.GetAudioConvertJob;
import com.volcengine.tos.TOSV2;
import com.volcengine.tos.TOSV2ClientBuilder;
import com.volcengine.tos.TosClientException;
import com.volcengine.tos.TosServerException;
import com.volcengine.tos.comm.common.*;
import com.volcengine.tos.model.bucket.*;
import com.volcengine.tos.model.object.*;
public class GetAudioConvertJobExample {
public static void main(String[] args) {
String endpoint = System.getenv("TOS_ENDPOINT");
String region = System.getenv("TOS_REGION");
String accessKey = System.getenv("TOS_ACCESS_KEY");
String secretKey = System.getenv("TOS_SECRET_KEY");
// 填写 BucketName，对应存放任务配置和结果的桶名称
String bucketName = System.getenv("TOS_BUCKET");
// 从环境变量中读取音频转码任务 ID，通常由创建任务接口返回
String jobId = "1885ff636fc5bb4331c40001834f292e698d";
// 初始化客户端
TOSV2 tos = new TOSV2ClientBuilder().build(region, endpoint, accessKey, secretKey);try {
// 构造查询音频转码任务的输入参数
// bucket 为创建任务时使用的桶名称
// jobId 为创建任务后返回的任务 ID，用于标识具体的音频转码任务
GetAudioConvertJobInput input = GetAudioConvertJobInput.builder()
 .bucket(bucketName)
 .jobId(jobId)
 .build();
// 调用查询音频转码任务接口
GetAudioConvertJobOutput output = tos.getAudioConvertJob(input);
// 打印关键返回信息：请求 ID、状态码、任务 ID、任务状态、错误码与错误信息
 com.volcengine.tos.model.RequestInfo requestInfo = output.getRequestInfo();
if (requestInfo != null) {
 System.out.println("查询音频转码任务成功");
 System.out.println("RequestID: " + requestInfo.getRequestId());
 System.out.println("StatusCode: " + requestInfo.getStatusCode());
 }
 System.out.println("JobID: " + output.getJobID());
 System.out.println("State: " + output.getState());
 System.out.println("Code: " + output.getCode());
 System.out.println("Message: " + output.getMessage());
// 若返回包含任务输入信息，则打印任务输入对象 Object
ConvertJobInput jobInput = output.getInput();
if (jobInput != null) {
 System.out.println("\n任务输入:");
// Object 表示待处理的源音频对象 Key
 System.out.println(" Object: " + jobInput.getObject());
 }
// 若返回包含任务输出信息，则打印输出对象所在的 Region、Bucket 与 Object
ConvertJobOutput jobOutput = output.getOutput();
if (jobOutput != null) {
 System.out.println("\n任务输出:");
// Region/Bucket/Object 分别表示转码结果输出所在的区域、桶与对象 Key
 System.out.println(" Region: " + jobOutput.getRegion());
 System.out.println(" Bucket: " + jobOutput.getBucket());
 System.out.println(" Object: " + jobOutput.getObject());
 }
// 若返回包含音频转码配置，则打印音频转码配置及时间区间
AudioConvertConfig config = output.getAudioConvertConfig();
if (config != null) {
 System.out.println("\n音频转码配置 AudioConvertConfig:");
// AudioConvertConfig 表示本次任务的音频转码参数配置
 System.out.println(" 容器格式 ContainerFormat: " + config.getContainerFormat());
 System.out.println(" 码率 BitRate: " + config.getBitRate() + " bps");
 System.out.println(" 采样格式 SampleFormat: " + config.getSampleFormat());
 System.out.println(" 采样率 SampleRate: " + config.getSampleRate() + " Hz");
 System.out.println(" 声道数 Channels: " + config.getChannels());
// TimeInterval 表示音频剪辑的起始时间与持续时长
TimeInterval timeInterval = config.getTimeInterval();

if (timeInterval != null) {
 System.out.println(" 时间区间 TimeInterval:");
 System.out.println(" 起始时间 Start: " + timeInterval.getStart() + " 毫秒");
 System.out.println(" 持续时间 Duration: " + timeInterval.getDuration() + " 毫秒");
 }
 }
 } catch (TosClientException e) {
// 操作失败，捕获客户端异常，一般情况是请求参数错误，此时请求并未发送
 System.out.println("操作失败");
 System.out.println("Message: " + e.getMessage());
if (e.getCause() != null) {
 e.getCause().printStackTrace();
 }
 } catch (TosServerException e) {
// 操作失败，捕获服务端异常，可以获取到从服务端返回的详细错误信息
 System.out.println("操作失败");
 System.out.println("StatusCode: " + e.getStatusCode());
 System.out.println("Code: " + e.getCode());
 System.out.println("Message: " + e.getMessage());
 System.out.println("RequestID: " + e.getRequestID());
 } catch (Throwable t) {
// 作为兜底捕获其他异常，一般不会执行到这里
 System.out.println("操作失败");
 System.out.println("unexpected exception, message: " + t.getMessage());
 }
 }
}
