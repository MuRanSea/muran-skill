9.2.8.5.3 删除水印模板（Java SDK）
本文介绍如何通过 TOS Java SDK 删除水印模板。
注意事项
仅 Java SDK 2.9.15 及以上版本删除水印模板的功能。
示例代码
import com.volcengine.tos.TOSV2;
import com.volcengine.tos.TOSV2ClientBuilder;
import com.volcengine.tos.TosClientException;
import com.volcengine.tos.TosServerException;
import com.volcengine.tos.model.bucket.DeleteWatermarkTemplateInput;
import com.volcengine.tos.model.bucket.DeleteWatermarkTemplateOutput;
public class DeleteWatermarkTemplateExample {
public static void main(String[] args) {
String endpoint = System.getenv("TOS_ENDPOINT");
String region = System.getenv("TOS_REGION");
String accessKey = System.getenv("TOS_ACCESS_KEY");
String secretKey = System.getenv("TOS_SECRET_KEY");
String bucketName = System.getenv("TOS_BUCKET");
String templateId = System.getenv("TOS_WATERMARK_TEMPLATE_ID");
TOSV2 tos = new TOSV2ClientBuilder().build(region, endpoint, accessKey, secretKey);try {
DeleteWatermarkTemplateOutput output =
tos.deleteWatermarkTemplate(DeleteWatermarkTemplateInput.builder()
 .bucket(bucketName)
 .id(templateId)
 .build());
 System.out.println("deleteWatermarkTemplate succeed.");
 System.out.println("Status code: " + output.getRequestInfo().getStatusCode());
 System.out.println("Request ID: " + output.getRequestInfo().getRequestId());
 } catch (TosClientException e) {
 System.out.println("deleteWatermarkTemplate failed");
 System.out.println("Message: " + e.getMessage());
if (e.getCause() != null) {
 e.getCause().printStackTrace();
 }
 } catch (TosServerException e) {
 System.out.println("deleteWatermarkTemplate failed");
