7.6.7 tosfsspec 生态集成
tosfsspec 完整实现了 fsspec 规范，除可直接在 Python 中使用外，还能与 Pandas 、 Dask 、 Ray 、PyArrow 、 PyTorch 等主流数据分析与 AI 框架无缝集成。
说明
传入构造参数前，您需要先配置长期访问凭证，具体操作请参见操作步骤。
在 fsspec 中使用
注意
之所以需要使用 tos-new:// 而不是使用 tos:// ，是因为当前非官网发布的tosfs 已在 fsspec 开源库注册，占用了 tos:// 这个协议头。
您可直接通过 fsspec.open() 访问 TOS 资源：在 URL 中使用 tos-new:// 协议，并通过 storage_options 传入所需的构造参数即可。
代码示例
fsspec.register_implementation 用法下， fsspec 会自动选择 TosFileSystem 作为底层文件系统实现。import fsspec
import os
import tos
fsspec.register_implementation('tos-new', 'tosfsspec.TosFileSystem')
url = 'tos-new://your-bucket/path/to/data.csv'
# 从环境变量获取 AKSK 信息
key = os.getenv('TOS_ACCESS_KEY')
secret = os.getenv('TOS_SECRET_KEY')
storage_options = {
'key': key, 
'secret': secret,
'region': 'cn-beijing'
}
with fsspec.open(url, 'rb', **storage_options) as f:
 head = f.read(1024)
print(head)

在 Python 中使用
在 Python 中可直接通过 tosfsspec 的常用 API 读写 TOS，类似本地文件系统的使用方式。代码示例
import os
import tosfsspec
import tos
# 从环境变量获取 AKSK 信息
key = os.getenv('TOS_ACCESS_KEY')
secret = os.getenv('TOS_SECRET_KEY')
fs = tosfsspec.TosFileSystem(
 storage_options = {
'key': key, 
'secret': secret,
'region': 'cn-beijing'
}
 region='cn-beijing'
)
bucket = 'your-bucket'
local_file_path = 'localfile.txt'
remote_file_path = f'tos://{bucket}/remote_file.txt'
# create a local file to upload
with open(local_file_path, 'w') as f:
 f.write('Hello TOSFS.')
# upload to tos
fs.put_file(local_file_path, remote_file_path)
print(f'Uploaded {local_file_path} to {remote_file_path}')
# download from tos
downloaded_file_path = 'downloaded_file.txt'
fs.get_file(remote_file_path, downloaded_file_path)
print(f'Downloaded {remote_file_path} to {downloaded_file_path}')
# read content from downloaded local file
with open(downloaded_file_path, 'r') as f:
 content = f.read()
print(f'Content of {downloaded_file_path}: {content}')
# delete tos file

与 PyArrow 集成
pyarrow 可通过 filesystem 参数使用 tosfsspec，实现高效读写 TOS 中的数据。
代码示例
import pyarrow as pa
import pyarrow.dataset as ds
import pyarrow.parquet as pq
import tosfsspec
import os
import tos
bucket = 'your-bucket'
# 从环境变量获取 AKSK 信息
key = os.getenv('TOS_ACCESS_KEY')
secret = os.getenv('TOS_SECRET_KEY')
fs = tosfsspec.TosFileSystem(
 key=key, 
 secret=secret,
 region='cn-beijing'
)
# 生成 Parquet 数据集
table = pa.table({'year': [2020, 2022, 2021, 2022, 2019, 2021],
'n_legs': [2, 2, 4, 4, 5, 100],
'animal': ['Flamingo', 'Parrot', 'Dog', 'Horse',
'Brittle stars', 'Centipede']})
pq.write_table(table, f'tos://{bucket}/data.parquet', filesystem=fs)
# 读取 Parquet 数据集
dataset = ds.dataset(f'tos://{bucket}/data.parquet', filesystem=fs)
# 输出数据集信息
print(f'Rows: {dataset.count_rows()}')
print(dataset.schema)
# 转换为 Pandas DataFrame
df = dataset.to_table().to_pandas()
print(df.head())

与 PyTorch 集成
PyTorch 可以在数据集加载和 Checkpoint 读写等场景集成 tosfsspec。
代码示例
import fsspec
import torch
import tosfsspec
from torchdata.datapipes.iter import FSSpecFileLister, FSSpecFileOpener
import os
import tos
bucket = 'your-bucket'
# 从环境变量获取 AKSK 信息
key = os.getenv('TOS_ACCESS_KEY')
secret = os.getenv('TOS_SECRET_KEY')
# 数据集加载
fsspec.register_implementation('tos-new', 'tosfsspec.TosFileSystem', )
storage_options = {
'key': key, 
'secret': secret,
'region': 'cn-beijing'
}
file_lister = FSSpecFileLister(root=f'tos-new://{bucket}/dataset/', **storage_options)iterable_dataset = FSSpecFileOpener(file_lister, mode='rb', **storage_options)
for _, item in iterable_dataset:
 data = item.read()
# Checkpoint 读写
fs = tosfsspec.TosFileSystem(
 key=key, 
 secret=secret,
 region='cn-beijing'
)
model = torch.nn.Linear(10, 10)
with fs.open(f'tos://{bucket}/models/model.pt', 'wb') as f:
 torch.save(model.state_dict(), f)
with fs.open(f'tos://{bucket}/models/model.pt', 'rb') as f:
 state_dict = torch.load(f)
# print(state_dict)
 model.load_state_dict(state_dict)
