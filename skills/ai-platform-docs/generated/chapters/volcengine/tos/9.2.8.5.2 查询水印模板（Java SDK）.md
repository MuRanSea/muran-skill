9.2.8.5.2 查询水印模板（Java SDK）
本文介绍如何通过 TOS Java SDK 查询水印模板。
注意事项
仅 Java SDK 2.9.15 及以上版本支持查询水印模板的功能。
删除操作不可恢复。执行删除前，请确认该模板 ID 归属于当前业务流程，避免误删其他任务正在使用的模板。
示例代码
import com.volcengine.tos.TOSV2;
import com.volcengine.tos.TOSV2ClientBuilder;
import com.volcengine.tos.TosClientException;
import com.volcengine.tos.TosServerException;
import com.volcengine.tos.model.bucket.ListWatermarkTemplatesInput;
import com.volcengine.tos.model.bucket.ListWatermarkTemplatesOutput;
import com.volcengine.tos.model.bucket.WatermarkTemplate;
import java.util.List;
public class ListWatermarkTemplatesExample {
public static void main(String[] args) {
String endpoint = System.getenv("TOS_ENDPOINT");
String region = System.getenv("TOS_REGION");
String accessKey = System.getenv("TOS_ACCESS_KEY");
String secretKey = System.getenv("TOS_SECRET_KEY");
String bucketName = System.getenv("TOS_BUCKET");
TOSV2 tos = new TOSV2ClientBuilder().build(region, endpoint, accessKey, secretKey);try {
ListWatermarkTemplatesOutput output =
tos.listWatermarkTemplates(ListWatermarkTemplatesInput.builder()
 .bucket(bucketName)
 .build());
List<WatermarkTemplate> templates = output.getWatermarkTemplates();
 System.out.println("listWatermarkTemplates succeed.");
 System.out.println("Watermark template count: " + templates.size());
for (WatermarkTemplate template : templates) {
 System.out.printf("Template ID: %s, name: %s, tag: %s.%n",
 template.getId(), template.getName(), template.getTag());
 }
 System.out.println("Request ID: " + output.getRequestInfo().getRequestId());

 } catch (TosClientException e) {
 System.out.println("listWatermarkTemplates failed");
 System.out.println("Message: " + e.getMessage());
if (e.getCause() != null) {
 e.getCause().printStackTrace();
 }
 } catch (TosServerException e) {
 System.out.println("listWatermarkTemplates failed");
 System.out.println("StatusCode: " + e.getStatusCode());
 System.out.println("Code: " + e.getCode());
 System.out.println("Message: " + e.getMessage());
 System.out.println("RequestID: " + e.getRequestID());
 } catch (Throwable t) {
 System.out.println("listWatermarkTemplates failed");
 System.out.println("unexpected exception, message: " + t.getMessage());
 }
 }
}
