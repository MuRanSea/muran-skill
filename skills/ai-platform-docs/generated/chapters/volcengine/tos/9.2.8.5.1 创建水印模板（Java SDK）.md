9.2.8.5.1 创建水印模板（Java SDK）
本文介绍如何通过 TOS Java SDK 创建水印模板。
注意事项
仅 Java SDK 2.9.15 及以上版本支持创建水印模板的功能。
示例代码
import com.volcengine.tos.TOSV2;
import com.volcengine.tos.TOSV2ClientBuilder;
import com.volcengine.tos.TosClientException;
import com.volcengine.tos.TosServerException;
import com.volcengine.tos.model.bucket.PutWatermarkTemplateInput;
import com.volcengine.tos.model.bucket.PutWatermarkTemplateOutput;
import com.volcengine.tos.model.bucket.Watermark;
import com.volcengine.tos.model.bucket.WatermarkText;
public class PutWatermarkTemplateExample {
public static void main(String[] args) {
String endpoint = System.getenv("TOS_ENDPOINT");
String region = System.getenv("TOS_REGION");
String accessKey = System.getenv("TOS_ACCESS_KEY");
String secretKey = System.getenv("TOS_SECRET_KEY");
String bucketName = System.getenv("TOS_BUCKET");
TOSV2 tos = new TOSV2ClientBuilder().build(region, endpoint, accessKey, secretKey);try {
WatermarkText watermarkText = WatermarkText.builder()
 .fontSize(50)
 .fontType("wqy-microhei")
 .fontColor("white")
 .transparency(20)
 .text("Java SDK Example")
 .build();
Watermark watermark = Watermark.builder()
 .type("Text")
 .pos("TopRight")
 .locMode("Relative")
 .dx(10)
 .dy(10)
 .startTime(0)

关于创建水印模板的相关 API 接口，请参见 PutVideoWatermarkTemplate 。
 .endTime(10000)
 .text(watermarkText)
 .build();
String templateName = "java-sdk-example-put-watermark-" + System.currentTimeMillis();PutWatermarkTemplateOutput output =
tos.putWatermarkTemplate(PutWatermarkTemplateInput.builder()
 .bucket(bucketName)
 .name(templateName)
 .watermarkConfig(watermark)
 .build());
 System.out.println("putWatermarkTemplate succeed.");
 System.out.println("Template name: " + templateName);
 System.out.println("Template ID: " + output.getId());
 System.out.println("Request ID: " + output.getRequestInfo().getRequestId());
 } catch (TosClientException e) {
 System.out.println("putWatermarkTemplate failed");
 System.out.println("Message: " + e.getMessage());
if (e.getCause() != null) {
 e.getCause().printStackTrace();
 }
 } catch (TosServerException e) {
 System.out.println("putWatermarkTemplate failed");
 System.out.println("StatusCode: " + e.getStatusCode());
 System.out.println("Code: " + e.getCode());
 System.out.println("Message: " + e.getMessage());
 System.out.println("RequestID: " + e.getRequestID());
 } catch (Throwable t) {
 System.out.println("putWatermarkTemplate failed");
 System.out.println("unexpected exception, message: " + t.getMessage());
 }
 }
}
