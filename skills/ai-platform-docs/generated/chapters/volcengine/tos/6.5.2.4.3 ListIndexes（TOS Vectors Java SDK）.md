6.5.2.4.3 ListIndexes（TOS Vectors Java SDK）
LlistIndexes 接口用于列出已创建的全部索引，本文介绍如何通过 TOS Vectors Java SDK 的 ListIndexes 接口获取索引列表。
注意事项
您需要根据实际情况，填写示例代码中的地域和访问域名，详细介绍，请参见 地域和访问域名（Endpoint）。
为了防止密钥泄露，本文示例代码需要从环境变量中读取 Access Key ID 和 Secret Access Key。因此使用 SDK 前，您需要配置长期访问凭证，详细介绍，请参见 配置长期访问凭证。
TOS Vectors Java SDK 支持使用临时访问凭证；也支持配置超时机制、重试策略等多种机制，您可以根据业务需求灵活设置，提升应用安全性，详细介绍，请参见 常见配置场景。
权限说明
主账号默认拥有全部权限，但是 IAM 用户默认没有任何权限。如果您需要使用 IAM 用户执行 ListIndexes操作，主账号需通过 IAM策略和桶策略为该 IAM 用户授予以下权限： tosvectors:ListIndexes代码示例
列举向量索引
package org.example.ListIndexes;
import com.volcengine.tos.TOSVectorsClientBuilder;
import com.volcengine.tos.TOSVectors;
import com.volcengine.tos.TosClientException;
import com.volcengine.tos.TosServerException;
import com.volcengine.tos.model.vectors.IndexSummary;
import com.volcengine.tos.model.vectors.ListIndexesInput;
import com.volcengine.tos.model.vectors.ListIndexesOutput;
public class ListIndexesExample {
public static void main(String[] args) {
String endpoint = System.getenv("TOS_ENDPOINT");
String region = System.getenv("TOS_REGION");
String accessKey = System.getenv("TOS_ACCESS_KEY");
String secretKey = System.getenv("TOS_SECRET_KEY");
String accountID = System.getenv("TOS_ACCOUNT_ID");
// 填写向量存储桶名称和索引名称
String vectorBucketName = System.getenv("TOS_VECTOR_BUCKET");
// 初始化客户端
TOSVectors client = new TOSVectorsClientBuilder().build(region, endpoint, accessKey, secretKey);try {
// 构建获取索引信息的输入参数
ListIndexesInput input = ListIndexesInput.builder()
 .vectorBucketName(vectorBucketName) // 设置向量存储桶名称
 .accountId(accountID) // 设置账户ID
 .build();
// 调用listIndexes API获取索引信息
ListIndexesOutput output = client.listIndexes(input);
// 打印请求信息
 System.out.println("RequestID: " + output.getRequestInfo().getRequestId());
 System.out.println("StatusCode: " + output.getRequestInfo().getStatusCode());
// 打印索引详细信息
if (output.getIndexes() != null) {
for (IndexSummary indexInfo : output.getIndexes()) {
 System.out.println("Index Name: " + indexInfo.getIndexName());
 System.out.println("Vector Bucket Name: " + indexInfo.getVectorBucketName());
 System.out.println("Creation Time: " + indexInfo.getCreationTime());
 System.out.println("Index TRN: " + indexInfo.getIndexTrn());
 }
 }
 System.out.println("获取向量索引信息成功");
 } catch (TosClientException e) {
// 操作失败，捕获客户端异常，一般情况是请求参数错误，此时请求并未发送
 System.out.println("操作失败");
 System.out.println("Message: " + e.getMessage());
if (e.getCause() != null) {
 e.getCause().printStackTrace();
 }
 } catch (TosServerException e) {
// 操作失败，捕获服务端异常，可以获取到从服务端返回的详细错误信息
 System.out.println("操作失败");
 System.out.println("StatusCode: " + e.getStatusCode());
 System.out.println("Code: " + e.getCode());
 System.out.println("Message: " + e.getMessage());
 System.out.println("RequestID: " + e.getRequestID());
 } catch (Throwable t) {
// 作为兜底捕获其他异常，一般不会执行到这里
 System.out.println("操作失败");
 System.out.println("unexpected exception, message: " + t.getMessage());
 }
 }
}
