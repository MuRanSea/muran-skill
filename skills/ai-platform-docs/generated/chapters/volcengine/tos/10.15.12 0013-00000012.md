问题原因
用户设置静态网站配置时，重定向规则中同时存在 ReplaceKeyPrefixWith 配置和 ReplaceKeyWith 配置。问题示例
您进行静态网站配置时，重定向规则中同时存在 ReplaceKeyPrefixWith 配置和 ReplaceKeyWith 配置。PUT /?website HTTP/1.1
Host: bucketname.tos-cn-beijing.volces.com
Date: Fri, 30 Jul 2021 13:59:18 GMT

"Protocol": "https"
 },
"IndexDocument": {
"Suffix": "index.html",
"ForbiddenSubDir": "false"
 },
"ErrorDocument": {
"Key": "error.html"
 },
"RoutingRules": [
 {
"Condition": {
"KeyPrefixEquals": "abc",
"HttpErrorCodeReturnedEquals": 403
 },
"Redirect": {
"HostName": "example.com",
"HttpRedirectCode": 301,
"Protocol": "http",
"ReplaceKeyPrefixWith": "redirect/",
"ReplaceKeyWith": "redirect.html"
 }
 }
