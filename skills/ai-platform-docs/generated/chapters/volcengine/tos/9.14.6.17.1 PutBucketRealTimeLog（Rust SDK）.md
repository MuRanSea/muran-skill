9.14.6.17 日志分析
9.14.6.17.1 PutBucketRealTimeLog（Rust SDK）
put_bucket_real_time_log 接口用于开启存储桶实时日志分析功能，本文介绍如何通过 Rust SDK 的put_bucket_real_time_log 接口配置 TLS 日志收集、角色授权及保留周期。
注意事项
您需要根据实际情况，填写示例代码中的地域和访问域名，详细介绍，请参见 地域和访问域名。为了防止密钥泄露，本文示例代码需要从环境变量中读取 Access Key ID 和 Secret Access Key。因此使用 SDK 前，您需要配置长期访问凭证，详细介绍，请参见 配置长期访问凭证。
use_service_topic 仅支持 true ，传入其他值会导致请求失败。
ttl 取值范围为 0 到 3650，超出范围将返回参数错误。
role 必须填写 TOSLogArchiveTLSRole ，使用其他角色名称会触发配置拒绝。
权限说明
火山引擎主账号默认拥有全部权限，但是 IAM 用户默认没有任何权限。如果您需要使用子账号执行put_bucket_real_time_log 操作，火山引擎主账号需通过 IAM策略和桶策略为该子账号授予以下权限：tos:PutBucketRealTimeLog

设置存储桶实时日志配置
use async_trait::async_trait;
use futures_core::future::BoxFuture;
use std::env;
use std::future::Future;
use std::time::Duration;
use tokio::runtime::Handle;
use ve_tos_rust_sdk::common::init_tracing_log;
use ve_tos_rust_sdk::asynchronous::bucket::BucketAPI;
use ve_tos_rust_sdk::asynchronous::tos;
use ve_tos_rust_sdk::asynchronous::tos::AsyncRuntime;
use ve_tos_rust_sdk::bucket::{PutBucketRealTimeLogInput, RealTimeLogConfiguration, AccessLogConfiguration};
use ve_tos_rust_sdk::error::TosError;
#[derive(Debug, Default)]
pub struct TokioRuntime {}
#[async_trait]
impl AsyncRuntime for TokioRuntime {
type JoinError = tokio::task::JoinError;
async fn sleep(&self, duration: Duration) {
 tokio::time::sleep(duration).await;
 }
fn spawn<'a, F>(&self, future: F) -> BoxFuture<'a, Result<F::Output, Self::JoinError>>where
 F: Future + Send + 'static,
 F::Output: Send + 'static,
 {
Box::pin(Handle::current().spawn(future))
 }
fn block_on<F: Future>(&self, future: F) -> F::Output {
 Handle::current().block_on(future)
 }
}
#[tokio::main]
async fn main() {
let _guard = init_tracing_log("info", "temp/logs", "app.log");
let ak = env::var("TOS_ACCESS_KEY").unwrap_or("".to_string());
let sk = env::var("TOS_SECRET_KEY").unwrap_or("".to_string());
let endpoint = "https://tos-cn-beijing.volces.com";
let region = "cn-beijing";
let client = tos::builder::<TokioRuntime>()
 .connection_timeout(3000)
 .request_timeout(10000)
 .max_retry_count(3)
 .ak(ak)
 .sk(sk)
 .region(region)
 .endpoint(endpoint)
 .build().unwrap();
// 设置存储桶实时日志配置
let mut access_log_config = AccessLogConfiguration::new();
 access_log_config.set_use_service_topic(true);
 access_log_config.set_tls_project_id("your-tls-project-id");
 access_log_config.set_tls_topic_id("your-tls-topic-id");
 access_log_config.set_tls_dashboard_id("your-tls-dashboard-id");
 access_log_config.set_ttl(7); // 日志保存7天
let real_time_log_config = RealTimeLogConfiguration::new("your-role", access_log_config);let input = PutBucketRealTimeLogInput::new("example-bucket", real_time_log_config);let output = client.put_bucket_real_time_log(&input).await;

 }
 _ => println!("未知错误: {:?}", err),
 }
 }
 }
