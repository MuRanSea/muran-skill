6.5.2.4 向量索引
6.5.2.4.1 GetIndex（TOS Vectors Java SDK）
GetIndex 接口用于获取指定索引的元数据与配置信息，本文介绍如何通过 GetIndex 接口查询索引结构、分片数、映射等属性。
注意事项
您需要根据实际情况，填写示例代码中的地域和访问域名，详细介绍，请参见 地域和访问域名（Endpoint）。
为了防止密钥泄露，本文示例代码需要从环境变量中读取 Access Key ID 和 Secret Access Key。因此使用 SDK 前，您需要配置长期访问凭证，详细介绍，请参见 配置长期访问凭证。
支持使用临时访问凭证；也支持配置超时机制、重试策略等多种机制，您可以根据业务需求灵活设置，提升应用安全性，详细介绍，请参见 常见配置场景。
权限说明
主账号默认拥有全部权限，但是 IAM 用户默认没有任何权限。如果您需要使用 IAM 用户执行 GetIndex 操作，主账号需通过 IAM策略和桶策略为该 IAM 用户授予以下权限： tosvectors:GetIndex
代码示例
获取指定向量索引的信息
// 获取指定向量索引的信息
import com.volcengine.tos.TOSVectorsClientBuilder;
import com.volcengine.tos.TOSVectors;
import com.volcengine.tos.TosClientException;
import com.volcengine.tos.TosServerException;
import com.volcengine.tos.model.vectors.GetIndexInput;
import com.volcengine.tos.model.vectors.GetIndexOutput;
public class GetIndexExample {
public static void main(String[] args) {
String endpoint = "Provide your endpoint";
String region = "Provide your region";
String accessKey = System.getenv("TOS_ACCESS_KEY");
String secretKey = System.getenv("TOS_SECRET_KEY");
String accountID = System.getenv("TOS_ACCOUNT_ID");
// 填写向量存储桶名称和索引名称
String vectorBucketName = "Provide your vector bucket name";
String indexName = "testindex";
// 初始化客户端
TOSVectors client = new TOSVectorsClientBuilder().build(region, endpoint, accessKey, secretKey);try {
// 构建获取索引信息的输入参数
GetIndexInput input = GetIndexInput.builder()
 .vectorBucketName(vectorBucketName) // 设置向量存储桶名称
 .indexName(indexName) // 设置索引名称
 .accountId(accountID) // 设置账户ID
 .build();
// 调用getIndex API获取索引信息
GetIndexOutput output = client.getIndex(input);
// 打印请求信息
 System.out.println("RequestID: " + output.getRequestInfo().getRequestId());
 System.out.println("StatusCode: " + output.getRequestInfo().getStatusCode());
// 打印索引详细信息
if (output.getIndex() != null) {
 System.out.println("Index Name: " + output.getIndex().getIndexName());
 System.out.println("Vector Bucket Name: " + output.getIndex().getVectorBucketName());
 System.out.println("Creation Time: " + output.getIndex().getCreationTime());
 System.out.println("Data Type: " + output.getIndex().getDataType());
 System.out.println("Dimension: " + output.getIndex().getDimension());
 System.out.println("Distance Metric: " + output.getIndex().getDistanceMetric());
 System.out.println("Index TRN: " + output.getIndex().getIndexTrn());
// 打印元数据配置信息
if (output.getIndex().getMetadataConfiguration() != null && 
 output.getIndex().getMetadataConfiguration().getNonFilterableMetadataKeys() != null) {
 System.out.println("Non-filterable Metadata Keys: " + 
 output.getIndex().getMetadataConfiguration().getNonFilterableMetadataKeys());
 }
 }
 System.out.println("获取向量索引信息成功");
 } catch (TosClientException e) {
// 操作失败，捕获客户端异常，一般情况是请求参数错误，此时请求并未发送
 System.out.println("操作失败");
 System.out.println("Message: " + e.getMessage());
if (e.getCause() != null) {
 e.getCause().printStackTrace();
 }
 } catch (TosServerException e) {
// 操作失败，捕获服务端异常，可以获取到从服务端返回的详细错误信息
 System.out.println("操作失败");
 System.out.println("StatusCode: " + e.getStatusCode());
 System.out.println("Code: " + e.getCode());
 System.out.println("Message: " + e.getMessage());
 System.out.println("RequestID: " + e.getRequestID());
 } catch (Throwable t) {
// 作为兜底捕获其他异常，一般不会执行到这里
 System.out.println("操作失败");
 System.out.println("unexpected exception, message: " + t.getMessage());
 }
 }
}
