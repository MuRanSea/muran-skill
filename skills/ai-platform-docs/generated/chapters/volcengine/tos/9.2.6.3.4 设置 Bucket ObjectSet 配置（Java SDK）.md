9.2.6.3.4 设置 Bucket ObjectSet 配置（Java SDK）
本文介绍如何通过 TOS Java SDK 设置或更新指定 Bucket 的 ObjectSet 配置，包括 ObjectSet 前缀层级、分隔符、默认 ObjectSet 开关、默认 QoS 和默认容量配额。
注意事项
使用前请确保目标 Bucket 已存在，并已初始化 TOSV2 客户端。
通过子账号调用时，需要授予当前 Bucket 的 tos:PutBucketObjectSetConfiguration 权限。更新配置时，仅请求中携带的字段会被序列化并提交；未设置的可选字段不会覆盖服务端已有配置。代码示例
以下示例为指定 Bucket 设置 ObjectSet 配置：将 ObjectSet 固定前缀层级设置为 3，使用 / 作为分隔符，启用默认 ObjectSet，并将默认 QoS 与容量配额设置为不限制。
import com.volcengine.tos.TOSV2;
import com.volcengine.tos.TOSV2ClientBuilder;
import com.volcengine.tos.TosClientException;
import com.volcengine.tos.TosServerException;
import com.volcengine.tos.model.bucket.ObjectSetQos;
import com.volcengine.tos.model.bucket.PutBucketObjectSetConfigurationInput;
import com.volcengine.tos.model.bucket.PutBucketObjectSetConfigurationOutput;
public class PutBucketObjectSetConfigurationExample {
public static void main(String[] args) {
String endpoint = System.getenv("TOS_ENDPOINT");
String region = System.getenv("TOS_REGION");
String accessKey = System.getenv("TOS_ACCESS_KEY");
String secretKey = System.getenv("TOS_SECRET_KEY");
String bucket = System.getenv("TOS_BUCKET");
TOSV2 tos = new TOSV2ClientBuilder().build(region, endpoint, accessKey, secretKey);try {
ObjectSetQos qos = ObjectSetQos.builder()
 .readsQps(0)
 .writesQps(0)
 .listQps(0)
 .readsRate(0)
 .writesRate(0)
 .build();
PutBucketObjectSetConfigurationInput input =
PutBucketObjectSetConfigurationInput.builder()
 .bucket(bucket)

 .pathLevel(3)
 .customDelimiter("/")
 .enableDefaultObjectSet(true)
 .qos(qos)
 .storageQuota("0")
 .build();

tos.putBucketObjectSetConfiguration(input);
 System.out.println("putBucketObjectSetConfiguration succeed");

 } catch (Throwable t) {

 }
}
