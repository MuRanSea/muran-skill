6.5.3.3.6 GetVectorBucketPolicy（TOS Vectors Node.js SDK）
GetVectorBucketPolicy 接口用于查询向量桶的 Policy，本文介绍如何通过 TOS Vectors Node.js SDK 的GetVectorBucketPolicy 接口完成该操作。
注意事项
您需要根据实际情况，填写示例代码中的地域和访问域名，详细介绍，请参见地域和访问域名（Endpoint）。
为了防止密钥泄露，本文示例代码需要从环境变量中读取 Access Key ID 和 Secret Access Key。因此使用 SDK 前，您需要配置长期访问凭证，详细介绍，请参见配置长期访问凭证。
TOS Vectors Node.js SDK 支持使用临时访问凭证；也支持配置超时机制、重试策略等多种机制，您可以根据业务需求灵活设置，提升应用安全性，详细介绍，请参见常见配置场景。
权限说明
主账号默认拥有全部权限，但是 IAM 用户默认没有任何权限。如果您需要使用 IAM 用户执行GetVectorBucketPolicy 操作，主账号需通过 IAM策略和桶策略为该 IAM 用户授予以下权限：tosvectors:GetVectorBucketPolicy
代码示例
查询向量桶的 Policy
// 查询向量桶的 Policy
import {
TosVectorsClient,
TosClientError,
TosServerError,
} from "@volcengine/tos-sdk";
// 从环境变量中获取认证信息和配置
const client = new TosVectorsClient({
accessKeyId: process.env["TOS_ACCESS_KEY"], // TOS_ACCESS_KEY: 您的 Access Key IDaccessKeySecret: process.env["TOS_SECRET_KEY"], // TOS_SECRET_KEY: 您的 Secret Keyregion: 'Provide your region',
endpoint: 'Provide your vector endpoint',
});
// 读取向量桶名称和账号 ID
const vectorBucketName = 'Provide your vector bucket name'; 
const accountId = process.env["TOS_ACCOUNT_ID"]; // TOS_ACCOUNT_ID: 您的账号 ID
/**
 * 处理 SDK 调用过程中可能出现的错误。
 * @param error 捕获到的错误对象
 */
function handleError(error: Error) {
if (error instanceof TosClientError) {
// 客户端异常，通常是参数错误或网络问题
console.log("Client Error:");
console.log(" Message:", error.message);
if (error.stack) {
console.log(" Stack:", error.stack);
 }
 } else if (error instanceof TosServerError) {
// 服务端异常，表示请求已送达但处理失败
console.log("Server Error:");
console.log(" Request ID:", error.requestId);
console.log(" Status Code:", error.statusCode);
console.log(" Code:", error.code);
console.log(" Message:", error.message);
console.log(" Headers:", error.headers);
 } else {
// 其他未知异常
console.log("Unexpected Error:");
console.log(" Message:", error.message);
 }
}
async function main() {
try {
// 2. 调用 getVectorBucketPolicy 接口
const result = await client.getVectorBucketPolicy({
 vectorBucketName,
accountId: accountId,
 });
// 3. 打印通用响应信息
console.log("请求已成功返回：");
console.log(" Request ID:", result.requestId);
console.log(" Status Code:", result.statusCode);
console.log(" Headers:", result.headers);
// 4. 打印策略详情
const policy = result.data;
if (!policy) {
console.log("未获取到任何 Policy 数据。");
return;
 }
console.log("策略基本信息：");
console.log(" Version:", policy.Version);
const statements = policy.Statement || [];
if (!Array.isArray(statements) || statements.length === 0) {
console.log(" 当前 Policy 不包含任何 Statement。");
return;
 }
console.log("策略语句列表（Statements）:");
 statements.forEach((statement: any, index: number) => {
console.log(` [Statement ${index + 1}]`);
if (statement.Sid !== undefined) {
console.log(" Sid:", statement.Sid);
 }
console.log(" Effect:", statement.Effect);
const actions = statement.Action;
if (Array.isArray(actions)) {
console.log(" Action:", actions.join(", "));
 } else if (actions !== undefined) {
console.log(" Action:", actions);
 }
const principal = statement.Principal;
if (Array.isArray(principal)) {
console.log(" Principal:", principal.join(", "));
 } else if (
 principal !== null &&
 principal !== undefined &&
typeof principal === "object"
 ) {
console.log(" Principal:", JSON.stringify(principal, null, 2));
 } else if (principal !== undefined) {
console.log(" Principal:", principal);
 }
const resource = statement.Resource;
if (Array.isArray(resource)) {
console.log(" Resource:", resource.join(", "));
 } else if (resource !== undefined) {
console.log(" Resource:", resource);
 }
 });
 } catch (error: any) {
handleError(error);
 }
}
main();
