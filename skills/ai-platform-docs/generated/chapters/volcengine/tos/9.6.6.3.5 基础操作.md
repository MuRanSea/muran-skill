9.6.6.3.5 基础操作
创建 ObjectSet（ Go SDK）
本文介绍如何通过 TOS Go SDK 在已开启 ObjectSet 的 Bucket 中创建 ObjectSet，并可同时写入 ObjectSet标签。
注意事项
Bucket 必须已开启 ObjectSet，可先调用 PutBucketObjectSetConfiguration 。
ObjectSetName 必须满足 Bucket 的分隔符和 PathLevel 配置。
TagSet 为可选参数，每个 ObjectSet 最多可配置 10 个标签。
通过子账号调用时，需要具备当前 Bucket 的 tos:PutObjectSet 权限。
示例代码
package main
import (
"context"
"fmt"
"os"
"github.com/volcengine/ve-tos-golang-sdk/v2/tos"
)
func checkErr(err error) {
if err != nil {
if serverErr, ok := err.(*tos.TosServerError); ok {
 fmt.Println("Error:", serverErr.Error())
 fmt.Println("Request ID:", serverErr.RequestID)
 fmt.Println("Response Status Code:", serverErr.StatusCode)
 fmt.Println("Response Header:", serverErr.Header)
 fmt.Println("Response Err Code:", serverErr.Code)
 fmt.Println("Response Err Msg:", serverErr.Message)
 } else if clientErr, ok := err.(*tos.TosClientError); ok {
 fmt.Println("Error:", clientErr.Error())
 fmt.Println("Client Cause Err:", clientErr.Cause.Error())
 } else {
 fmt.Println("Error:", err)
 }
panic(err)
 }
}

 Bucket: bucketName,
 ObjectSetName: objectSetName,

 {Key: "env", Value: "docs"},
 }},
 }
 putOutput, err := client.PutObjectSet(ctx, input)
 checkErr(err)
 fmt.Println("PutObjectSet Request ID:", putOutput.RequestID)
}
