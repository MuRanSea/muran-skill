9.2.8.2.5 列举视频转码模板（Java SDK）
本文介绍如何通过 TOS Java SDK 列举视频转码模板。
注意事项
仅 Java SDK 2.9.8 及以上版本支持列举视频转码模板的功能。
示例代码
// 列举视频转码模板
package org.example.ListVideoConvertTemplates;
import com.volcengine.tos.TOSV2;
import com.volcengine.tos.TOSV2ClientBuilder;
import com.volcengine.tos.TosClientException;
import com.volcengine.tos.TosServerException;
import com.volcengine.tos.comm.common.*;
import com.volcengine.tos.model.bucket.*;
import com.volcengine.tos.model.object.*;
public class ListVideoConvertTemplatesExample {
public static void main(String[] args) {
String endpoint = System.getenv("TOS_ENDPOINT");
String region = System.getenv("TOS_REGION");
String accessKey = System.getenv("TOS_ACCESS_KEY");
String secretKey = System.getenv("TOS_SECRET_KEY");
// 填写 BucketName，对应要列举视频转码模板所在的桶
String bucketName = System.getenv("TOS_BUCKET");
// 初始化客户端，使用环境变量中的 endpoint、region、AK/SK 进行鉴权
TOSV2 tos = new TOSV2ClientBuilder().build(region, endpoint, accessKey, secretKey);try {
// 创建列举视频转码模板请求，只需要指定 Bucket 名称
ListVideoConvertTemplatesInput input = ListVideoConvertTemplatesInput.builder()
 .bucket(bucketName)
 .build();
// 调用列举视频转码模板接口
ListVideoConvertTemplatesOutput output = tos.listVideoConvertTemplates(input);
// 打印本次请求的关键信息，便于排查问题
 System.out.println("列举视频转码模板成功");
if (output.getRequestInfo() != null) {
 System.out.println("RequestID: " + output.getRequestInfo().getRequestId());
 System.out.println("StatusCode: " + output.getRequestInfo().getStatusCode());
 }
// 获取视频转码模板列表
 java.util.List<VideoConvertTemplate> templates = output.getVideoConvertTemplates();if (templates != null && !templates.isEmpty()) {
 System.out.println("找到 " + templates.size() + " 个视频转码模板:");
for (VideoConvertTemplate template : templates) {
 System.out.println("\n模板信息:");
 System.out.println(" ID: " + template.getId());
 System.out.println(" 名称: " + template.getName());
 System.out.println(" 标签: " + template.getTag());
// 打印转码配置 Transcode 的核心字段
Transcode transcodeConfig = template.getTranscodeConfig();
if (transcodeConfig != null) {
 System.out.println(" 转码配置:");
// 容器配置，例如 MP4/TS 以及切片时长
Container container = transcodeConfig.getContainer();
if (container != null) {
 System.out.println(" 容器配置:");
 System.out.println(" 容器格式: " + container.getFormat());
ClipConfig clipConfig = container.getClipConfig();
if (clipConfig != null) {
 System.out.println(" 切片时长: " + clipConfig.getDuration());
 }
 }
// 视频参数，如分辨率、码率、帧率等
Video video = transcodeConfig.getVideo();
if (video != null) {
 System.out.println(" 视频参数:");
 System.out.println(" 编码格式: " + video.getCodec());
 System.out.println(" 分辨率: " + video.getWidth() + "x" + video.getHeight());
 System.out.println(" 码率: " + video.getBitRate());
 System.out.println(" 帧率: " + video.getFps());
 System.out.println(" 像素格式: " + video.getPixFmt());
 System.out.println(" 是否移除视频流: " + video.getRemove());
 }
// 音频参数，如码率、采样率、声道数等
Audio audio = transcodeConfig.getAudio();
if (audio != null) {

 System.out.println(" 编码格式: " + audio.getCodec());
 System.out.println(" 码率: " + audio.getBitRate());
 System.out.println(" 采样率: " + audio.getSampleRate());
 System.out.println(" 声道数: " + audio.getChannels());

 }
 } else {
 System.out.println("未找到视频转码模板");
 }
 } catch (TosClientException e) {
// 操作失败，捕获客户端异常，一般情况是请求参数错误，此时请求并未发送
 System.out.println("操作失败");
 System.out.println("Message: " + e.getMessage());
if (e.getCause() != null) {
 e.getCause().printStackTrace();
 }
 } catch (TosServerException e) {
// 操作失败，捕获服务端异常，可以获取到从服务端返回的详细错误信息
 System.out.println("操作失败");
 System.out.println("StatusCode: " + e.getStatusCode());
 System.out.println("Code: " + e.getCode());
 System.out.println("Message: " + e.getMessage());
 System.out.println("RequestID: " + e.getRequestID());
 } catch (Throwable t) {
// 作为兜底捕获其他异常，一般不会执行到这里
 System.out.println("操作失败");
 System.out.println("unexpected exception, message: " + t.getMessage());
 }
 }
}
