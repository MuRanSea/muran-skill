6.5.3.5.4 QueryVectors（TOS Vectors Node.js SDK）
QueryVectors 接口用于查询距离最近TopK向量，本文介绍如何通过 TOS Vectors Node.js SDK 的QueryVectors 接口完成该操作。
注意事项
您需要根据实际情况，填写示例代码中的地域和访问域名，详细介绍，请参见地域和访问域名（Endpoint）。
为了防止密钥泄露，本文示例代码需要从环境变量中读取 Access Key ID 和 Secret Access Key。因此使用 SDK 前，您需要配置长期访问凭证，详细介绍，请参见配置长期访问凭证。
TOS Vectors Node.js SDK 支持使用临时访问凭证；也支持配置超时机制、重试策略等多种机制，您可以根据业务需求灵活设置，提升应用安全性，详细介绍，请参见常见配置场景。
权限说明
主账号默认拥有全部权限，但是 IAM 用户默认没有任何权限。如果您需要使用 IAM 用户执行QueryVectors 操作，主账号需通过 IAM策略和桶策略为该 IAM 用户授予以下权限：
tosvectors:QueryVectors
代码示例
查询距离最近 TopK 向量
// 查询距离最近TopK向量
import {
TosVectorsClient,
TosClientError,
TosServerError,
} from "@volcengine/tos-sdk";
// 从环境变量中获取认证信息和配置
const client = new TosVectorsClient({
accessKeyId: process.env["TOS_ACCESS_KEY"], // TOS_ACCESS_KEY: 您的 Access Key IDaccessKeySecret: process.env["TOS_SECRET_KEY"], // TOS_SECRET_KEY: 您的 Secret Keyregion: 'Provide your region',
endpoint: 'Provide your vector endpoint',
});
const vectorBucketName = 'Provide your vector bucket name';
const accountId = process.env["TOS_ACCOUNT_ID"];
const indexName = "test";
/**
 * 处理 SDK 调用过程中可能出现的错误。
 * @param error 捕获到的错误对象
 */
function handleError(error: Error) {
if (error instanceof TosClientError) {
// 客户端异常，通常是参数错误或网络问题
console.log("Client Error:");
console.log(" Message:", error.message);
if (error.stack) {
console.log(" Stack:", error.stack);
 }
 } else if (error instanceof TosServerError) {
// 服务端异常，表示请求已送达但处理失败
console.log("Server Error:");
console.log(" Request ID:", error.requestId);
console.log(" Status Code:", error.statusCode);
console.log(" Code:", error.code);
console.log(" Message:", error.message);
console.log(" Headers:", error.headers);
 } else {
// 其他未知异常
console.log("Unexpected Error:");
console.log(" Message:", error.message);
 }
}
async function main() {
// 请确保以下维度与创建索引时的 dimension 保持一致
const dimension = 128;
const topK = 5;
try {
console.log(
`Querying top ${topK} nearest vectors from index: ${indexName} in vector bucket: $
{vectorBucketName}...`,
 );
// 1. 构造查询向量，与索引维度一致
const queryVectorData = Array.from({ length: dimension }, () =>
Math.random(),
 );
// 2. 组装 QueryVectors 输入参数
// 3. 调用 queryVectors 方法查询最近邻向量
const result = await client.queryVectors({
 vectorBucketName,
accountID: accountId,
 indexName,
returnDistance: true,
returnMetadata: true,
 topK,
queryVector: {
float32: queryVectorData,
 },
// 如需按业务元数据过滤，可在此处设置 filter，例如：
// filter: { category: "book" },
 });
// 4. 打印通用返回信息
console.log("QueryVectors Request Succeeded:");
console.log(" Request ID:", result.requestId);
console.log(" Status Code:", result.statusCode);
console.log(" Headers:", result.headers);
// 5. 打印业务数据核心字段：返回的向量 key、distance 和 metadata
const vectors = result.data?.vectors;
if (!vectors || vectors.length === 0) {
console.log(" No vectors returned for this query.");
return;
 }
console.log(" Matched Vectors:");
 vectors.forEach((vector, index) => {
console.log(` [${index}] Key: ${vector.key}`);
console.log(` Distance: ${vector.distance}`);
console.log(" Metadata:", vector.metadata);
 });
 } catch (error: any) {
handleError(error);
 }
}
main();
