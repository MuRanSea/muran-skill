9.2.8.6.2 提交视频转动图任务（Java SDK）
本文介绍如何通过 TOS Java SDK 提交视频转动图任务。
注意事项
仅 Java SDK 2.9.15 及以上版本支持提交视频转动图任务的功能。
创建接口只返回任务 ID。任务状态、输出对象、动图配置等详情需使用 GetVideoAnimationJob 查询。
示例代码
以下代码展示如何对已有视频对象提交转 GIF 任务，并将输出文件保存到同一桶下。

import com.volcengine.tos.TOSV2ClientBuilder;

import com.volcengine.tos.model.bucket.CreateAsyncProcessTaskInput;
import com.volcengine.tos.model.bucket.CreateAsyncProcessTaskOutput;
import java.nio.charset.StandardCharsets;
import java.util.Base64;

public static void main(String[] args) {
String endpoint = System.getenv("TOS_ENDPOINT");
String region = System.getenv("TOS_REGION");
String accessKey = System.getenv("TOS_ACCESS_KEY");
String secretKey = System.getenv("TOS_SECRET_KEY");
String bucketName = System.getenv("TOS_BUCKET");
String objectKey = System.getenv("TOS_OBJECT_KEY");
TOSV2 tos = new TOSV2ClientBuilder().build(region, endpoint, accessKey, secretKey);try {
String outputKey = "java-sdk-example/create-async-process-task/output-"
 + System.currentTimeMillis() + ".gif";
String saveObject =
Base64.getEncoder().encodeToString(outputKey.getBytes(StandardCharsets.UTF_8));
String asyncProcess = "x-tos-async-process=video/
animation,w_0,h_400,ss_0,f_gif,fps_10,num_10,inter_200"
 + "&x-tos-save-object=" + saveObject;
CreateAsyncProcessTaskOutput output =
tos.createAsyncProcessTask(CreateAsyncProcessTaskInput.builder()
 .bucket(bucketName)
