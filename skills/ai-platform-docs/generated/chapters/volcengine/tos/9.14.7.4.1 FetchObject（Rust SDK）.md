9.14.7.4 抓取对象
9.14.7.4.1 FetchObject（Rust SDK）
fetch_object 接口用于从公开 URL 抓取单个大小不超过 5 GiB 的对象并写入指定存储桶，支持加密、ACL与存储类型设置。本文介绍如何通过 Rust SDK 的 fetch_object 接口完成抓取。
注意事项
您需要根据实际情况，填写示例代码中的地域和访问域名，详细介绍，请参见 地域和访问域名。为了防止密钥泄露，本文示例代码需要从环境变量中读取 Access Key ID 和 Secret Access Key。因此使用 SDK 前，您需要配置长期访问凭证，详细介绍，请参见 配置长期访问凭证。
单次调用仅抓取 1 个对象，对象大小上限 5 GiB，超出后抓取失败。
url 参数仅接受可公开访问的地址，私有地址返回错误。
若设置 storage_class ，目的对象存储类型以该值为准；未设置则继承桶默认类型。
使用 SSE_C 加密时， ssec_algorithm 、 ssec_key 、 ssec_key_md5 必须全部提供。
权限说明
火山引擎主账号默认拥有全部权限，但是 IAM 用户默认没有任何权限。如果您需要使用子账号执行fetch_object 操作，火山引擎主账号需通过 IAM策略和桶策略为该子账号授予以下权限：
tos:PutObject

use async_trait::async_trait;
use futures_core::future::BoxFuture;
use std::env;
use std::future::Future;
use std::time::Duration;
use tokio::runtime::Handle;
use ve_tos_rust_sdk::common::init_tracing_log;
use ve_tos_rust_sdk::asynchronous::object::{ObjectAPI};
use ve_tos_rust_sdk::asynchronous::tos;
use ve_tos_rust_sdk::asynchronous::tos::AsyncRuntime;
use ve_tos_rust_sdk::error::TosError;
use ve_tos_rust_sdk::object::{FetchObjectInput};
#[derive(Debug, Default)]
pub struct TokioRuntime {}
#[async_trait]
impl AsyncRuntime for TokioRuntime {
type JoinError = tokio::task::JoinError;
async fn sleep(&self, duration: Duration) {
 tokio::time::sleep(duration).await;
 }
fn spawn<'a, F>(&self, future: F) -> BoxFuture<'a, Result<F::Output, Self::JoinError>>where
 F: Future + Send + 'static,
 F::Output: Send + 'static,
 {
Box::pin(Handle::current().spawn(future))
 }
fn block_on<F: Future>(&self, future: F) -> F::Output {
 Handle::current().block_on(future)
 }
}
#[tokio::main]
async fn main() {
// Initialize logging
let _guard = init_tracing_log("info", "temp/logs", "app.log");
// Get credentials from environment variables
let ak = env::var("TOS_ACCESS_KEY").unwrap_or("".to_string());
let sk = env::var("TOS_SECRET_KEY").unwrap_or("".to_string());
let endpoint = "https://tos-cn-beijing.volces.com";
let region = "cn-beijing";
// Build TOS client
let client = tos::builder::<TokioRuntime>()
 .connection_timeout(3000)
 .request_timeout(10000)
 .max_retry_count(3)
 .ak(ak)
 .sk(sk)
 .region(region)
 .endpoint(endpoint)
 .build()
 .expect("Failed to build TOS client");
// Prepare FetchObject input
let mut input = FetchObjectInput::new("example-bucket", "target-object-key");
 input.set_url("https://example.com/public-source-file.txt"); // Required public URL
 input.set_ignore_same_key(true); // Optional: skip if target exists
// Execute fetch object operation
相关文档
关于 FetchObject 的 API 文档，请参见 FetchObject。
let result = client.fetch_object(&input).await;
// Handle result
match result {
Ok(output) => {
println!("Fetch successful!");
println!("ETag: {}", output.etag());
println!("Source Content Length: {}", output.source_content_length());
println!("Source Content Type: {}", output.source_content_type());
println!("MD5: {}", output.md5());
 },
Err(err) => {
match err {
 TosError::TosClientError { message, .. } => {
println!("Client error: {}", message);
 },
 TosError::TosServerError { request_id, ec, code, .. } => {
println!("Server error: Request ID: {}, EC: {}, Code: {}", request_id, ec, code);
 },
 }
 },
 }
}
