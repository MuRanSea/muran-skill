9.14.7.1.3 DeleteMultiObjects（Rust SDK）
delete_multi_objects 接口用于批量删除桶内对象，每次最多 1000 条。本文介绍如何通过 Rust SDK 的delete_multi_objects 接口完成批量删除，支持版本控制与静默模式。
注意事项
您需要根据实际情况，填写示例代码中的地域和访问域名，详细介绍，请参见 地域和访问域名。为了防止密钥泄露，本文示例代码需要从环境变量中读取 Access Key ID 和 Secret Access Key。因此使用 SDK 前，您需要配置长期访问凭证，详细介绍，请参见 配置长期访问凭证。
单次调用最多可删除 1000 条对象，超出需分批。
若桶已开启版本控制且未提供 version_Id ，系统会插入 delete_marker 而非永久删除。
recursive 与 skip_trash 仅在开启回收站的分层命名空间生效，普通桶会被忽略。
quiet 为 true 时，响应仅返回删除失败的条目，无条目表示全部成功。
权限说明
火山引擎主账号默认拥有全部权限，但是 IAM 用户默认没有任何权限。如果您需要使用子账号执行delete_multi_objects 操作，火山引擎主账号需通过 IAM策略和桶策略为该子账号授予以下权限：tos:DeleteMultiObjects
代码示例
批量删除对象
use async_trait::async_trait;
use futures_core::future::BoxFuture;
use std::env;
use std::future::Future;
use std::time::Duration;
use tokio::runtime::Handle;
use ve_tos_rust_sdk::common::init_tracing_log;
use ve_tos_rust_sdk::asynchronous::bucket::BucketAPI;
use ve_tos_rust_sdk::asynchronous::object::ObjectAPI;
use ve_tos_rust_sdk::asynchronous::tos;
use ve_tos_rust_sdk::asynchronous::tos::AsyncRuntime;
use ve_tos_rust_sdk::object::{DeleteMultiObjectsInput, ObjectTobeDeleted};
use ve_tos_rust_sdk::error::TosError;
#[derive(Debug, Default)]
pub struct TokioRuntime {}
#[async_trait]
impl AsyncRuntime for TokioRuntime {
type JoinError = tokio::task::JoinError;
async fn sleep(&self, duration: Duration) {
 tokio::time::sleep(duration).await;
 }
fn spawn<'a, F>(&self, future: F) -> BoxFuture<'a, Result<F::Output, Self::JoinError>>where
 F: Future + Send + 'static,
 F::Output: Send + 'static,
 {
Box::pin(Handle::current().spawn(future))
 }
fn block_on<F: Future>(&self, future: F) -> F::Output {
 Handle::current().block_on(future)
 }
}
#[tokio::main]
async fn main() {
let _guard = init_tracing_log("info", "temp/logs", "app.log");
let ak = env::var("TOS_ACCESS_KEY").unwrap_or("".to_string());
let sk = env::var("TOS_SECRET_KEY").unwrap_or("".to_string());
let endpoint = "https://tos-cn-beijing.volces.com";
let region = "cn-beijing";
let client = tos::builder::<TokioRuntime>()
 .connection_timeout(3000)
 .request_timeout(10000)
 .max_retry_count(3)
 .ak(ak)
 .sk(sk)
 .region(region)
 .endpoint(endpoint)
 .build().unwrap();
// Batch delete objects
let bucket = "example-bucket";
let objects = vec![
 ObjectTobeDeleted::new("obj1"),
 ObjectTobeDeleted::new("obj2"),
 ObjectTobeDeleted::new("obj3"),
 ];
let input = DeleteMultiObjectsInput::new_with_objects(bucket, objects);
let output = client.delete_multi_objects(&input).await;
match output {
Ok(output) => {
println!("Deleted objects:");
批量删除指定版本的对象
for (index, deleted) in output.deleted().iter().enumerate() {
println!("[{}] Key: {}, VersionId: {}, DeleteMarker: {}, DeleteMarkerVersionId: {}",
 index, deleted.key(), deleted.version_id(), deleted.delete_marker(), 
deleted.delete_marker_version_id());
 }
println!("Errors:");
for (index, error) in output.error().iter().enumerate() {
println!("[{}] Key: {}, VersionId: {}, Code: {}, Message: {}",
 index, error.key(), error.version_id(), error.code(), error.message());
 }
 }
Err(err) => {
match err {
 TosError::TosClientError { message, .. } => {
println!("Client error: message: {}", message);
 }
 TosError::TosServerError { request_id, ec, code, .. } => {
println!("Server error: request_id: {}, ec: {}, code: {}", request_id, ec, code);
 }
 }
 }
 }
}
