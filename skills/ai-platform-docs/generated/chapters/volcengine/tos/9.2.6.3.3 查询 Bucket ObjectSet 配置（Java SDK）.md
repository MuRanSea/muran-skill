9.2.6.3.3 查询 Bucket ObjectSet 配置（Java SDK）
本文介绍如何通过 TOS Java SDK 查询 Bucket 级 ObjectSet 配置，包括前缀层级、自定义分隔符、默认ObjectSet 开关、默认 QoS 和默认容量配额。
注意事项
Bucket 未配置 ObjectSet 时，服务端可能返回 NoSuchBucketObjectSetConfiguration 。通过子账号调用时，需要具备当前 Bucket 的 tos:GetBucketObjectSetConfiguration 权限。代码示例
import com.volcengine.tos.TOSV2;
import com.volcengine.tos.TOSV2ClientBuilder;
import com.volcengine.tos.TosClientException;
import com.volcengine.tos.TosServerException;
import com.volcengine.tos.model.bucket.GetBucketObjectSetConfigurationInput;
import com.volcengine.tos.model.bucket.GetBucketObjectSetConfigurationOutput;
public class GetBucketObjectSetConfigurationExample {
public static void main(String[] args) {
String endpoint = System.getenv("TOS_ENDPOINT");
String region = System.getenv("TOS_REGION");
String accessKey = System.getenv("TOS_ACCESS_KEY");
String secretKey = System.getenv("TOS_SECRET_KEY");
String bucket = System.getenv("TOS_BUCKET");
TOSV2 tos = new TOSV2ClientBuilder().build(region, endpoint, accessKey, secretKey);try {
GetBucketObjectSetConfigurationInput input =
GetBucketObjectSetConfigurationInput.builder()
 .bucket(bucket)
 .build();
GetBucketObjectSetConfigurationOutput output =
tos.getBucketObjectSetConfiguration(input);
 System.out.println("getBucketObjectSetConfiguration succeed");
 System.out.println("PathLevel: " + output.getPathLevel());
 System.out.println("CustomDelimiter: " + output.getCustomDelimiter());
 System.out.println("EnableDefaultObjectSet: " + output.getEnableDefaultObjectSet());
 System.out.println("StorageQuota: " + output.getStorageQuota());
 System.out.println("RequestID: " + output.getRequestInfo().getRequestId());
 } catch (TosClientException e) {
 System.out.println("getBucketObjectSetConfiguration failed");
 System.out.println("Message: " + e.getMessage());
