9.6.6.3 ObjectSet
9.6.6.3.1 设置 Bucket ObjectSet 配置（ Go SDK）
本文介绍如何通过 TOS Go SDK 设置或更新 Bucket 的 ObjectSet 配置，包括固定前缀层级、自定义分隔符、默认 ObjectSet 开关、默认 QoS 与默认容量配额。
注意事项
目标 Bucket 必须已存在。
通过子账号调用时，需要具备当前 Bucket 的 tos:PutBucketObjectSetConfiguration 权限。示例代码
package main
import (
"context"
"fmt"
"os"
"github.com/volcengine/ve-tos-golang-sdk/v2/tos"
)
func checkErr(err error) {
if err != nil {
if serverErr, ok := err.(*tos.TosServerError); ok {
 fmt.Println("Error:", serverErr.Error())
 fmt.Println("Request ID:", serverErr.RequestID)
 fmt.Println("Response Status Code:", serverErr.StatusCode)
 fmt.Println("Response Header:", serverErr.Header)
 fmt.Println("Response Err Code:", serverErr.Code)
 fmt.Println("Response Err Msg:", serverErr.Message)
 } else if clientErr, ok := err.(*tos.TosClientError); ok {
 fmt.Println("Error:", clientErr.Error())
 fmt.Println("Client Cause Err:", clientErr.Cause.Error())
 } else {
 fmt.Println("Error:", err)
 }
panic(err)
 }
}

 input := &tos.PutBucketObjectSetConfigurationInput{
 Bucket: bucketName,
 PathLevel: 3,
 CustomDelimiter: "/",
 EnableDefaultObjectSet: true,

 ReadsQps: 0,
 WritesQps: 0,
 ListQps: 0,
 ReadsRate: 0,
 WritesRate: 0,
 },
 }
 putOutput, err := client.PutBucketObjectSetConfiguration(ctx, input)
 checkErr(err)
 fmt.Println("PutBucketObjectSetConfiguration Request ID:", putOutput.RequestID)
