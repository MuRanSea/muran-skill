 messages = [
 {"role": "user", "content": "Hello"},
 ],

"type": "disabled", # 不使用深度思考能力

from openai import OpenAI
import os
client = OpenAI(
# The base URL for model invocation
 base_url="https://ark.cn-beijing.volces.com/api/v3",
# Replace with your API Key
 api_key=os.environ.get("ARK_API_KEY"),
)
completion = client.chat.completions.create(
# Replace with Model ID
 model="doubao-seed-2-1-pro-260915",
 messages = [
 {"role": "user", "content": "Hello"},
 ],
# 自定义request id
 extra_headers={"X-Client-Request-Id": "202406251728190000B7EA7A9648AC08D9"}
)
print(completion.choices[0].message.content)

from langchain_openai import ChatOpenAI
from langchain_core.prompts import PromptTemplate
import os
llm = ChatOpenAI(
# Replace with your API Key
 openai_api_key=os.environ.get("ARK_API_KEY"),
# The base URL for model invocation
 openai_api_base="https://ark.cn-beijing.volces.com/api/v3",
# Replace with Model ID
 model="doubao-seed-2-1-pro-260915",
)
template = """Question: {question}
Answer: Let's think step by step."""
prompt = PromptTemplate.from_template(template)
question = "What NFL team won the Super Bowl in the year Justin Beiber was born?"llm_chain = prompt | llm
print(llm_chain.invoke(question))
